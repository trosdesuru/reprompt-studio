// frontend/features/admin/hooks/useAdminDashboard.ts
import { useState, useCallback } from "react";
import { api } from "@/src/services/api";
import { AuditLog } from "../admin.types";

export function useAdminDashboard() {
  const [inviteCode, setInviteCode] = useState<string | null>(null);
  const [isLoadingInvite, setIsLoadingInvite] = useState(false);

  const [logs, setLogs] = useState<AuditLog[]>([]);
  const [isLoadingLogs, setIsLoadingLogs] = useState(false);

  const fetchLogs = useCallback(async () => {
    setIsLoadingLogs(true);
    try {
      const data = await api.obtenerLogsAuditoria(20);
      if (data.status === "success") {
        setLogs(data.logs);
      }
    } catch (err) {
      console.error("Error cargando logs:", err);
    } finally {
      setIsLoadingLogs(false);
    }
  }, []);

  const handleGenerateInvite = async (role: string) => {
    setIsLoadingInvite(true);
    setInviteCode(null);
    try {
      const data = await api.generarInvitacion(role);
      setInviteCode(data.code);
    } catch (error: any) {
      alert(`❌ Error: ${error.message}`);
    } finally {
      setIsLoadingInvite(false);
    }
  };

  const handleRevokeAll = async () => {
    if (
      !confirm(
        "⚠️ ¿Estás seguro? Esto invalidará todas las sesiones en Redis y te expulsará incluso a ti.",
      )
    )
      return;
    try {
      const data = await api.revocarTodasLasSesiones();
      if (data.status === "success") {
        alert("🔒 Sesiones invalidadas con éxito.");
        window.location.href = "/login";
      }
    } catch (err) {
      alert("Error al comunicarse con Redis para revocar sesiones.");
    }
  };

  return {
    logs,
    isLoadingLogs,
    inviteCode,
    isLoadingInvite,
    fetchLogs,
    handleGenerateInvite,
    handleRevokeAll,
  };
}

export function useAdminGuard() {
  const { user, isLoadingAuth } = useAuthStore();
  const router = useRouter();

  useEffect(() => {
    if (!isLoadingAuth && (!user || user.role !== "admin")) {
      router.push("/");

"use client";

import { useEffect } from "react";
import { usePathname } from "next/navigation";
import { useAuthStore } from "@/src/store/authStore";

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const { checkAuth, user, hasCheckedInitialAuth } = useAuthStore();
  const pathname = usePathname() || "";

  useEffect(() => {
    const isAuthPage =
      pathname.includes("/login") || pathname.includes("/join") || pathname.includes("/callback");

    if (!isAuthPage) {
      if (!user && !hasCheckedInitialAuth) {
        checkAuth();
      }
    } else {
      useAuthStore.setState({ isLoadingAuth: false });
    }
  }, [pathname, checkAuth, user, hasCheckedInitialAuth]);

  return <>{children}</>;
}

// src/features/auth/hooks/useAuthForm.ts
import { useState, useEffect } from "react";
import { useSearchParams } from "next/navigation";
import { useOAuth } from "./useOAuth";
import { useCredentialsAuth } from "./useCredentialsAuth";

export function useAuthForm() {
  const searchParams = useSearchParams();

  // Dependencias
  const oauth = useOAuth();
  const credsAuth = useCredentialsAuth();

  // Estado puro de la UI (Inputs)
  const [isLogin, setIsLogin] = useState(true);
  const [emailOrUser, setEmailOrUser] = useState("");
  const [password, setPassword] = useState("");
  const [inviteCode, setInviteCode] = useState("");
  const [isInviteLocked, setIsInviteLocked] = useState(false);

  // Manejo de código de invitación por URL
  useEffect(() => {
    const codeFromUrl = searchParams.get("code");
    if (codeFromUrl) {
      setIsLogin(false);
      setInviteCode(codeFromUrl);
      setIsInviteLocked(true);
    }
  }, [searchParams]);

  // Wrappers para inyectar el estado actual en las funciones de los submódulos
  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    credsAuth.handleCredentialsSubmit(isLogin, emailOrUser, password, inviteCode);
  };

  const handleOauthSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    oauth.handleOauthSubmit(inviteCode);
  };

  // Estado derivado para bloquear la UI si ALGO está cargando
  const isAnyLoading = credsAuth.isCredsLoading || oauth.isOauthSubmitting;

  return {
    state: {
      isLogin,
      isLoading: isAnyLoading,
      isGoogleLoading: oauth.isGoogleLoading,
      isDiscordLoading: oauth.isDiscordLoading,
      emailOrUser,
      password,
      inviteCode,
      isInviteLocked,
      oauthPending: oauth.oauthPending,
    },
    actions: {
      setIsLogin,
      setEmailOrUser,
      setPassword,
      setInviteCode,
      handleSubmit,
      handleOauthSubmit,
      handleGoogleLogin: oauth.handleGoogleLogin,
      handleDiscordLogin: oauth.handleDiscordLogin,
      cancelOauthFlow: oauth.cancelOauthFlow,
    },
  };
}

// src/features/auth/hooks/useCredentialsAuth.ts
import { useState } from "react";
import { useRouter } from "next/navigation";
import { api } from "@/src/services/api";
import toast from "react-hot-toast";

export function useCredentialsAuth() {
  const router = useRouter();
  const [isCredsLoading, setIsCredsLoading] = useState(false);

  const handleCredentialsSubmit = async (
    isLogin: boolean,
    emailOrUser: string,
    password: string,
    inviteCode: string,
  ) => {
    setIsCredsLoading(true);
    try {
      if (isLogin) {
        await api.login({ username: emailOrUser, password });
        toast.success("¡Bienvenido de nuevo!");
        router.push("/es/studio");
      } else {
        await api.register({ email: emailOrUser, password, invite_code: inviteCode });
        toast.success("Cuenta creada con éxito");
        await api.login({ username: emailOrUser, password });
        router.push("/es/studio");
      }
    } catch (error: any) {
      toast.error(error?.response?.data?.detail || "Error en la autenticación");
    } finally {
      setIsCredsLoading(false);
    }
  };

  return { handleCredentialsSubmit, isCredsLoading };
}

export function useRedirectIfAuthenticated(redirectTo: string = "/") {
  const router = useRouter();
  const { user, isLoadingAuth } = useAuthStore();

  useEffect(() => {
    if (!isLoadingAuth && user) {
      router.replace(redirectTo);
    }
  }, [user, isLoadingAuth, router, redirectTo]);

    isChecking: isLoadingAuth,
    isAuthenticated: !!user,
  };
}

import { useState, useEffect } from "react";
import { useSearchParams, useRouter } from "next/navigation";
import { api } from "@/src/services/api";
import toast from "react-hot-toast";

export function useOAuthPending() {
  const router = useRouter();
  const searchParams = useSearchParams();

  const [oauthPending, setOauthPending] = useState<any>(null);
  const [isOauthSubmitting, setIsOauthSubmitting] = useState(false);

  useEffect(() => {
    const pendingData = sessionStorage.getItem("oauth_pending");
    if (pendingData) {
      setOauthPending(JSON.parse(pendingData));
    }
  }, []);

  const handleOauthSubmit = async (inviteCode: string) => {
    if (!oauthPending) return;
    setIsOauthSubmitting(true);
    try {
      await api.auth.registerGoogleUser({
        token: oauthPending.token,
        invite_code: inviteCode,
        username: oauthPending.username,
      });
      sessionStorage.removeItem("oauth_pending");
      toast.success("¡Cuenta VIP creada con éxito!");
      window.location.href = "/es/studio";
    } catch (error: any) {
      toast.error(error.message || "Código inválido");
    } finally {
      setIsOauthSubmitting(false);
    }
  };

  const cancelOauthFlow = () => {
    setOauthPending(null);
    sessionStorage.removeItem("oauth_pending");
    if (searchParams.get("step") === "oauth_invite") {
      router.replace("/es/login");
    }
  };

  return {
    oauthPending,
    isOauthSubmitting,
    handleOauthSubmit,
    cancelOauthFlow,
  };
}

import { useState, useCallback, useEffect, useRef } from "react";
import { parsePromptData, ExtractedPromptData } from "../utils/promptParser";
import { getRawTextFromFile } from "../utils/fileExtractors";
import { extractFilesFromZip } from "../utils/fileExtractors/zipExtractor";
import { handleError } from "@/src/utils/errors/ErrorHandler";
import { useAuthStore } from "@/src/store/authStore";

export interface BatchItem {
  id: string;
  file: File;
  previewUrl: string | null;
  data: ExtractedPromptData;
  status: "idle" | "saving" | "success" | "error";
}

export interface SkippedFile {
  name: string;
  reason: string;
}

export function usePromptExtractor() {
  const [isExtracting, setIsExtracting] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [batch, setBatch] = useState<BatchItem[]>([]);
  const [skippedFiles, setSkippedFiles] = useState<SkippedFile[]>([]);
  const [upgradeModal, setUpgradeModal] = useState(false);

  const { user } = useAuthStore();

  const activeUrls = useRef<Set<string>>(new Set());

  const closeUpgradeModal = useCallback(() => {
    setUpgradeModal(false);
  }, []);

  useEffect(() => {
    return () => {
      activeUrls.current.forEach((url) => {
        URL.revokeObjectURL(url);
        console.log(`🗑️ RAM liberada (Cleanup): ${url}`);
      });
      activeUrls.current.clear();
    };
  }, []);

  const processFiles = useCallback(
    async (files: File[], fileRejections: any[] = []) => {
      setUpgradeModal(false);
      setIsExtracting(true);
      setError(null);

      let allFilesToProcess: File[] = [];
      const currentSkipped: SkippedFile[] = fileRejections.map((r) => ({
        name: r.file.name,
        reason: "Formato no aceptado o archivo inválido.",
      }));

      const maxFilesAllowed = user?.role === "artist" ? 200 : 10;

      try {
        // 1. Fase de expansión (ZIPs) - Sigue siendo igual
        for (const file of files) {
          if (
            file.type === "application/zip" ||
            file.name.endsWith(".zip") ||
            file.name.endsWith(".x-zip-compressed")
          ) {
            const unzipped = await extractFilesFromZip(file, maxFilesAllowed);
            allFilesToProcess = [...allFilesToProcess, ...unzipped.validFiles];
            currentSkipped.push(...unzipped.rejectedFiles);
            if (unzipped.limitReached && user?.role === "voyager") setUpgradeModal(true);
          } else {
            if (allFilesToProcess.length < maxFilesAllowed) {
              allFilesToProcess.push(file);
            } else {
              currentSkipped.push({ name: file.name, reason: "Límite de subida alcanzado" });
              if (user?.role === "voyager") setUpgradeModal(true);
            }
          }
        }

        // 🚀 2. LA MAGIA: PROCESAMIENTO CONCURRENTE LIMITADO (2 a la vez)
        const newBatchItems: BatchItem[] = [];
        const queue = [...allFilesToProcess]; // Copiamos la lista de archivos

        // Definimos un "trabajador" que procesa la cola
        const worker = async () => {
          while (queue.length > 0) {
            const file = queue.shift(); // Sacamos el primer archivo de la lista
            if (!file) continue;

            try {
              // Extracción de texto (Esta es la parte pesada de CPU)
              const rawText = await getRawTextFromFile(file);
              let parsedData = rawText
                ? parsePromptData(rawText)
                : {
                    prompt: "",
                    rawText: "",
                    software: "Desconocido",
                    negativePrompt: "",
                    model: "",
                    loras: [],
                    extractionConfidence: 0,
                  };

              const previewUrl = file.type.startsWith("image/") ? URL.createObjectURL(file) : null;

              newBatchItems.push({
                id: crypto.randomUUID(),
                file,
                previewUrl,
                data: parsedData as ExtractedPromptData,
                status: "idle",
              });
            } catch (e) {
              currentSkipped.push({ name: file.name, reason: "Error al procesar el archivo." });
            }
          }
        };

        // Lanzamos exactamente 2 trabajadores.
        // El ventilador de tu Mac bajará de revoluciones al instante.
        await Promise.all([worker(), worker()]);

        // 3. Actualización de estado final
        setSkippedFiles((prev) => [...prev, ...currentSkipped]);
        if (newBatchItems.length > 0) setBatch((prev) => [...prev, ...newBatchItems]);
      } catch (err: unknown) {
        setError(handleError(err, "ExtractorBatch"));
      } finally {
        setIsExtracting(false);
      }
    },
    [user],
  );

  const updateItemStatus = useCallback((id: string, status: BatchItem["status"]) => {
    setBatch((prev) => prev.map((item) => (item.id === id ? { ...item, status } : item)));
  }, []);

  const updateItemData = useCallback((id: string, newData: Partial<ExtractedPromptData>) => {
    setBatch((prev) =>
      prev.map((item) => (item.id === id ? { ...item, data: { ...item.data, ...newData } } : item)),
    );
  }, []);

  const removeItem = useCallback((id: string) => {
    setBatch((prev) => {
      const item = prev.find((p) => p.id === id);
      if (item?.previewUrl) {
        URL.revokeObjectURL(item.previewUrl);
        activeUrls.current.delete(item.previewUrl);
      }
      return prev.filter((p) => p.id !== id);
    });
  }, []);

  const clearBatch = useCallback(() => {
    setBatch((prev) => {
      prev.forEach((item) => {
        if (item.previewUrl) {
          URL.revokeObjectURL(item.previewUrl);
          activeUrls.current.delete(item.previewUrl);
        }
      });
      return [];
    });
    setSkippedFiles([]);
    setError(null);
  }, []);

  const clearSkipped = useCallback(() => {
    setSkippedFiles([]);
  }, []);

  return {
    isExtracting,
    error,
    batch,
    skippedFiles,
    processFiles,
    upgradeModal,
    updateItemStatus,
    updateItemData,
    removeItem,
    clearBatch,
    clearSkipped,
    closeUpgradeModal,
  };
}

import { ExtractedPromptData, ExtractedPromptSchema } from "./parsers/validation";
import { cleanNullBytes } from "./parsers/stringUtils";
import { parseA1111 } from "./parsers/a1111Parser";
import { parseComfy } from "./parsers/comfyParser";
import { parseMidjourney } from "./parsers/midjourneyParser";

export type { ExtractedPromptData };

export function parsePromptData(rawText: string): ExtractedPromptData {
  const cleanText = cleanNullBytes(rawText);
  let parsed: Partial<ExtractedPromptData>;

  if (cleanText.trim().startsWith("{")) {
    try {
      const jsonObj = JSON.parse(cleanText);
      parsed = parseComfy(jsonObj, cleanText);
    } catch {
      parsed = parseA1111(cleanText);
    }
  } else if (/--ar|--v|--style|--niji|Job ID:/i.test(cleanText)) {
    parsed = parseMidjourney(cleanText, rawText);
    parsed.software = "Midjourney";
  } else {
    parsed = parseA1111(cleanText);
  }

  const validation = ExtractedPromptSchema.safeParse({
    ...parsed,
    rawText: rawText,
  });

  if (!validation.success) {
    console.error("❌ Fallo en contrato de extracción:", validation.error.format());

    return {
      prompt: cleanText,
      negativePrompt: "",
      resolution: undefined,
      rawText: rawText,
      software: "Desconocido",
      extractionConfidence: 0.1,
    };
  }

  return validation.data;
}

export async function getRawTextFromFile(file: File): Promise<string | null> {
  const type = file.type;

  if (type === "image/png") {
    return extractPngMetadata(file);
  }

  if (type === "image/jpeg" || type === "image/jpg") {
    return extractJpegMetadata(file);
  }

  if (type === "text/plain" || type === "application/json") {
    return extractRawText(file);
  }

  return null;
}

import exifr from "exifr";

export async function extractJpegMetadata(file: File): Promise<string | null> {
  try {
    // Escaneamos solo lo necesario para no sobrecargar
    const metadata = await exifr.parse(file, {
      userComment: true,
      xmp: true,
      iptc: true,
      // Nota: exifr busca en ImageDescription por defecto si no se filtra
    });

    if (!metadata) return null;

    // 1. Prioridad: UserComment (Comfy/Civitai/A1111)
    if (metadata.userComment) {
      const raw =
        typeof metadata.userComment === "string"
          ? metadata.userComment
          : new TextDecoder("utf-8").decode(metadata.userComment);

      return raw.replace(/^(UNICODE|ASCII)\x00+/, "").trim();
    }

    // 2. Prioridad: ImageDescription (Midjourney/A1111 alternativo)
    // exifr suele mapear 'Description' a 'ImageDescription'
    if (metadata.ImageDescription) {
      return metadata.ImageDescription.trim();
    }

    if (metadata.description) {
      if (typeof metadata.description === "string") {
        return metadata.description.trim();
      }
      if (Array.isArray(metadata.description)) {
        return metadata.description.join(" ").trim();
      }
      return String(metadata.description).trim();
    }

    return null;
  } catch (error) {
    console.error("❌ Error en jpegExtractor:", error);
    return null;
  }
}

import extract from "png-chunks-extract";
import text from "png-chunk-text";

export async function extractPngMetadata(file: File): Promise<string | null> {
  const buffer = await file.arrayBuffer();
  const chunks = extract(new Uint8Array(buffer));
  const textChunks = chunks.filter((c: any) => c.name === "tEXt" || c.name === "iTXt");

  for (const chunk of textChunks) {
    const decoded = text.decode(chunk.data);
    if (decoded.keyword === "parameters" || decoded.keyword === "Comment") {
      return decoded.text;
    }
  }
  return null;
}

import JSZip from "jszip";
import { string } from "zod";

export async function extractFilesFromZip(
  zipFile: File,
  maxFilesAllowed: number,
): Promise<ZipExtractionResult> {
  const zip = new JSZip();
  const loadedZip = await zip.loadAsync(zipFile);

  const validFiles: File[] = [];
  const rejectedFiles: { name: string; reason: string }[] = [];
  let limitReached = false;

  const supportedRegex = /\.(png|jpe?g|webp|txt|json)$/i;
  let fileCount = 0;

  for (const [filename, zipEntry] of Object.entries(loadedZip.files)) {
    if (
      zipEntry.dir ||
      filename.includes("__MACOSX") ||
      filename.split("/").pop()?.startsWith(".")
    ) {
      continue;
    }

    if (fileCount >= maxFilesAllowed) {
      rejectedFiles.push({
        name: filename,
        reason: `Requiere plan superior. Límite: ${maxFilesAllowed} archivos.`,
      });
      limitReached = true;
      continue;
    }

    if (!supportedRegex.test(filename)) {
      rejectedFiles.push({
        name: filename.split("/").pop() || filename,
        reason: "Formato no soportado",
      });
      continue;
    }

    try {
      const blob = await zipEntry.async("blob");

      if (blob.size > 50 * 1024 * 1024) {
        rejectedFiles.push({ name: filename, reason: "Archivo demasiado grande (>50MB)." });
        continue;
      }

      const mimeType = getMimeType(filename);
      const file = new File([blob], filename.split("/").pop() || filename, { type: mimeType });
      validFiles.push(file);
      fileCount++;
    } catch (e) {
      rejectedFiles.push({ name: filename, reason: "Archivo corrupto dentro del ZIP" });
    }
  }

  return { validFiles, rejectedFiles, limitReached };
}

function getMimeType(filename: string): string {
  const lower = filename.toLowerCase();
  if (lower.endsWith(".png")) return "image/png";
  if (lower.endsWith(".jpg") || lower.endsWith(".jpeg")) return "image/jpeg";
  if (lower.endsWith(".webp")) return "image/webp";
  if (lower.endsWith(".json")) return "application/json";
  return "text/plain";
}

import { ExtractedPromptData } from "./types";
import { cleanA1111Prefixes } from "./stringUtils";

export function parseA1111(rawText: string): ExtractedPromptData {
  const result: ExtractedPromptData = {
    prompt: "",
    rawText,
    software: "A1111",
  };

  const cleanText = cleanA1111Prefixes(rawText);

  const negativePromptRegex = /Negative prompt:\s*([\s\S]*?)(?=\nSteps:|$)/i;
  const negativeMatch = cleanText.match(negativePromptRegex);

  if (negativeMatch) {
    result.negativePrompt = negativeMatch[1].trim();
    result.prompt = cleanText.split(/Negative prompt:/i)[0].trim();
  } else {
    result.prompt = cleanText.split(/\nSteps:/i)[0].trim();
  }

  const matchers = {
    steps: /Steps:\s*([^,]+)/i,
    sampler: /Sampler:\s*([^,]+)/i,
    cfgScale: /CFG scale:\s*([^,]+)/i,
    seed: /Seed:\s*([^,]+)/i,
    resolution: /Size:\s*([^,]+)/i,
    model: /Model:\s*([^,]+)/i,
  };

  Object.entries(matchers).forEach(([key, regex]) => {
    const match = cleanText.match(regex);
    if (match) (result as any)[key] = match[1].trim();
  });

  return result;
}

import { ExtractedPromptData } from "./types";
import { extractFileName } from "./stringUtils";

export function parseComfy(jsonObj: any, rawText: string): ExtractedPromptData {
  const result: ExtractedPromptData = { prompt: "", rawText, software: "ComfyUI" };

  if (jsonObj.extraMetadata) {
    const meta =
      typeof jsonObj.extraMetadata === "string"
        ? JSON.parse(jsonObj.extraMetadata)
        : jsonObj.extraMetadata;

    result.prompt = meta.prompt || result.prompt;
    result.negativePrompt = meta.negativePrompt || "";
    result.steps = meta.steps?.toString();
    result.sampler = meta.sampler;
    result.cfgScale = meta.cfgScale?.toString();
    result.seed = meta.seed?.toString();

    if (meta.images?.[0]) result.resolution = `${meta.images[0].width}x${meta.images[0].height}`;

    if (meta.resources) {
      result.model = meta.resources
        .find((r: any) => r.type === "Checkpoint")
        ?.modelVersionId?.toString();
      result.loras = meta.resources
        .filter((r: any) => r.type === "LORA")
        .map((l: any) => ({ name: l.modelVersionId?.toString(), strength: l.strength }));
    }

    if (meta.triggerWords) {
      result.triggerWords = Array.isArray(meta.triggerWords)
        ? meta.triggerWords
            .flatMap((w: string) => w.split(","))
            .map((w: string) => w.trim())
            .filter(Boolean)
        : [];
    }

    if (result.prompt && result.steps) return result;
  }

  if (jsonObj && typeof jsonObj === "object" && !Array.isArray(jsonObj)) {
    Object.values(jsonObj).forEach((node: any) => {
      if (typeof node !== "object" || !node) return;

      if (node.class_type?.includes("KSampler")) {
        result.steps = node.inputs?.steps?.toString() || result.steps;
        result.seed = node.inputs?.seed?.toString() || result.seed;
        result.cfgScale = node.inputs?.cfg?.toString() || result.cfgScale;
        result.sampler = node.inputs?.sampler_name || result.sampler;
      }
      if (node.class_type?.includes("CLIPTextEncode")) {
        const text = node.inputs?.text;
        const isNeg = node._meta?.title?.toLowerCase().includes("negative");
        if (isNeg) result.negativePrompt = text;
        else if (!result.prompt) result.prompt = text;
      }
      if (node.class_type === "LoraLoader") {
        if (!result.loras) result.loras = [];
        result.loras.push({
          name: extractFileName(node.inputs?.lora_name),
          strength: node.inputs?.strength_model,
        });
      }
    });
  }

  return result;
}

import { ExtractedPromptData } from "./types";

export function parseMidjourney(description: string, rawText: string): ExtractedPromptData {
  const result: ExtractedPromptData = {
    prompt: description,
    rawText,
    software: "Desconocido",
  };

  if (description.includes("--") || description.includes("Job ID:")) {
    result.software = "Desconocido";

    const arMatch = description.match(/--ar\s+([0-9:]+)/);
    if (arMatch) result.resolution = arMatch[1];

    const noMatch = description.match(/--no\s+([^--]+)/);
    if (noMatch) {
      result.negativePrompt = noMatch[1].trim();
    }
  }

  return result;
}

import { z } from "zod";

export const ExtractedPromptSchema = z.object({
  prompt: z
    .string()
    .min(1, "El prompt es obligatorio")
    .transform((t) => t.trim()),
  negativePrompt: z.string().default(""),
  steps: z.string().optional(),
  sampler: z.string().optional(),
  cfgScale: z.string().optional(),
  seed: z.string().optional(),
  model: z.string().optional(),
  resolution: z
    .string()
    .optional()
    .transform((r) => r?.replace(/\s+/g, "")),
  software: z.enum(["A1111", "ComfyUI", "Midjourney", "NovelAI", "Desconocido"]),
  rawText: z.string(),
  extractionConfidence: z.number().min(0).max(1).default(1),

  loras: z
    .array(
      z.object({
        name: z.string(),
        strength: z.union([z.string(), z.number()]),
      }),
    )
    .optional(),
  triggerWords: z.array(z.string()).optional(),
});

export type ExtractedPromptData = z.infer<typeof ExtractedPromptSchema>;

import { FeedItem } from "@/src/types/models";
import { useState, useEffect, useCallback } from "react";
import { api } from "@/src/services/api";

export function usePublicFeed(query: string) {
  const [feed, setFeed] = useState<FeedItem[]>([]);
  const [isFeedLoading, setIsFeedLoading] = useState(false);
  const [feedError, setFeedError] = useState<string | null>(null);
  const [nextCursor, setNextCursor] = useState<string | null>(null);
  const [hasMore, setHasMore] = useState(true);

  const loadFeed = useCallback(
    async (isInitial: boolean = false) => {
      if (isFeedLoading || (!isInitial && !hasMore)) return;

      setIsFeedLoading(true);
      setFeedError(null);

      try {
        const currentCursor = isInitial ? null : nextCursor;
        const res = await api.public.getFeed({
          include_self: false,
          cursor: currentCursor,
        });

        const data = await res.json();

        if (data.status === "success") {
          setFeed((prev) => (isInitial ? data.feed : [...prev, ...data.feed]));
          setNextCursor(data.next_cursor);
          setHasMore(!!data.next_cursor);
        }
      } catch (err) {
        console.error("Error cargando feed:", err);
        setFeedError("Hubo un problema al cargar el arte.");
      } finally {
        setIsFeedLoading(false);
      }
    },
    [nextCursor, hasMore, isFeedLoading],
  );

  useEffect(() => {
    if (query) return;
    loadFeed(true);
  }, [query]);

  return {
    feed,
    isFeedLoading,
    feedError,
    hasMore,
    loadMore: () => loadFeed(false),
  };
}

// frontend/features/image-viewer/hooks/useViewerSync.ts
import { useState, useCallback } from "react";
import { api } from "@/src/services/api";
import { ViewerPageItem } from "@/src/types/models";

export function useViewerSync(activeUrl: string) {
  const [isViewerOpen, setIsViewerOpen] = useState(false);
  const [viewerPages, setViewerPages] = useState<ViewerPageItem[]>([]);
  const [isViewerLoading, setIsViewerLoading] = useState(false);
  const [isLoadingMore, setIsLoadingMore] = useState(false);
  const [hasMorePages, setHasMorePages] = useState(false);
  const [currentViewerPage, setCurrentViewerPage] = useState(0);

  const openViewer = async (targetUrl?: string) => {
    const urlToFetch = targetUrl || activeUrl;

    if (!urlToFetch) {
      console.warn("⚠️ [QA DEBUG] openViewer abortado: targetUrl es undefined");
      return;
    }

    setIsViewerOpen(true);
    setIsViewerLoading(true);
    setViewerPages([]);

    try {
      const res = await api.leer(urlToFetch, 0);
      const dataArray = res.data || res.paginas || res.enlaces;

      if (res.status === "success" || res.status === "ok" || (dataArray && dataArray.length > 0)) {
        if (Array.isArray(dataArray) && dataArray.length > 0) {
          console.log(`✅ [QA DEBUG] Guardando ${dataArray.length} imágenes en el estado.`);
          setViewerPages(dataArray);
          setHasMorePages(!!res.has_next);
          setCurrentViewerPage(1);
        } else {
          console.warn(
            "⚠️ [QA DEBUG] El Backend devolvió SUCCESS, pero el array de imágenes está VACÍO o no es un array válido.",
          );
          setViewerPages([]);
        }
      } else {
        console.error(
          "❌ [QA DEBUG] El Backend devolvió un status de ERROR o estructura irreconocible:",
          res,
        );
        setViewerPages([]);
      }
    } catch (err) {
      console.error("💥 [QA DEBUG - CAPA 2] Crash durante la petición de red:", err);
    } finally {
      setIsViewerLoading(false);
    }
  };

  const closeViewer = () => {
    setIsViewerOpen(false);
    setViewerPages([]);
  };

  const loadNextPage = useCallback(async () => {
    if (!hasMorePages || isLoadingMore) return;
    setIsLoadingMore(true);

    try {
      const res = await api.leer(activeUrl, currentViewerPage);
      const dataArray = res.data || res.paginas || res.enlaces;

      if (Array.isArray(dataArray)) {
        setViewerPages((prev) => [...prev, ...dataArray]);
        setHasMorePages(!!res.has_next);

        if (res.has_next) setCurrentViewerPage((prev) => prev + 1);
      }
    } catch (e) {
      console.error(e);
    } finally {
      setIsLoadingMore(false);
    }
  }, [activeUrl, currentViewerPage, hasMorePages, isLoadingMore]);

  return {
    viewerPages,
    isViewerLoading,
    isLoadingMore,
    isViewerOpen,
    openViewer,
    closeViewer,
    hasMorePages,
    loadNextPage,
  };
}

import { useState, useCallback } from "react";
import { fetchSeguro } from "@/src/services/api/client";

interface SearchOptions {
  scope: "global" | "studio";
  directoryId?: string;
}

export function useVectorSearch({ scope, directoryId }: SearchOptions) {
  const [query, setQuery] = useState("");
  const [results, setResults] = useState<any[]>([]);
  const [isLoading, setIsLoading] = useState(false);

  const handleExecute = useCallback(async () => {
    if (!query.trim()) return;

    setIsLoading(true);
    try {
      const params = new URLSearchParams({
        q: query,
        scope,
        ...(directoryId && { directory_id: directoryId }),
      });

      const res = await fetchSeguro(`/search/vector?${params.toString()}`);
      if (res.ok) {
        const data = await res.json();
        setResults(data.results || []);
      }
    } catch (err) {
      console.error(`[VectorSearch:${scope}] Error:`, err);
    } finally {
      setIsLoading(false);
    }
  }, [query, scope, directoryId]);

  return {
    query,
    setQuery,
    results,
    isLoading,
    handleExecute,
    clear: () => {
      setQuery("");
      setResults([]);
    },
  };
}

import { useCallback } from "react";
import { useStudioUI } from "@/src/store/useStudioUI";
import { fetchSeguro } from "@/src/services/api/client";

const API_URL = process.env.NEXT_PUBLIC_API_URL || "http://localhost:8000/api";

export function useDirectoryActions() {
  const { directories, setDirectories, activeDirectoryId, setActiveDirectory } = useStudioUI();

  const createDirectory = useCallback(
    async (name: string, parentId: string | null) => {
      try {
        const payload: Record<string, string> = { name };
        if (parentId) {
          payload.parent_id = parentId;
        }

        const res = await fetchSeguro(`${API_URL}/directories/`, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify(payload),
          credentials: "include",
        });

        if (res.ok) {
          const newDir = await res.json();
          setDirectories([...directories, newDir]);
        } else {
          throw new Error("Fallo al crear el directorio");
        }
      } catch (error) {
        console.error("[DirectoryActions] Error creando:", error);
      }
    },
    [directories, setDirectories],
  );

  // En src/features/studio/hooks/useDirectoryActions.ts

  const deleteDirectory = useCallback(
    async (id: string) => {
      try {
        const res = await fetchSeguro(`${API_URL}/directories/${id}`, {
          method: "DELETE",
          credentials: "include",
        });

        if (res.ok) {
          setDirectories(directories.filter((d) => d.id !== id));
          if (activeDirectoryId === id) setActiveDirectory(null);
        } else {
          // 🟢 FRONTEND INTELIGENTE: Leemos el error del backend
          const errorData = await res.json();

          if (res.status === 409 && errorData.detail === "DIRECTORY_NOT_EMPTY") {
            alert(
              "⚠️ Esta carpeta contiene subcarpetas o prompts.\n\nPor seguridad, debes vaciarla o mover sus elementos antes de eliminarla.",
            );
            return;
          }

          throw new Error(errorData.detail || "Fallo al borrar el directorio");
        }
      } catch (error: any) {
        console.error("[DirectoryActions] Error borrando:", error.message);
      }
    },
    [directories, setDirectories, activeDirectoryId, setActiveDirectory],
  );

  const renameDirectory = useCallback(
    async (id: string, newName: string) => {
      try {
        const res = await fetchSeguro(`${API_URL}/directories/${id}`, {
          method: "PATCH",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ name: newName }),
        });

        if (!res.ok) throw new Error("No se pudo renombrar");

        setDirectories(directories.map((dir) => (dir.id === id ? { ...dir, name: newName } : dir)));
      } catch (error) {
        console.error("[DirectoryActions] Error renombrando:", error);
        throw error;
      }
    },
    [directories, setDirectories],
  );

  const moveDirectory = useCallback(
    async (directoryId: string, newParentId: string | null) => {
      const previousDirectories = [...directories];
      setDirectories(
        directories.map((d) => (d.id === directoryId ? { ...d, parent_id: newParentId } : d)),
      );

      try {
        const res = await fetchSeguro(`${API_URL}/directories/${directoryId}/move`, {
          method: "PATCH",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ parent_id: newParentId }),
          credentials: "include",
        });

        if (!res.ok) throw new Error("Fallo al mover en el servidor");
      } catch (error) {
        console.error("[DirectoryActions] Error moviendo, haciendo rollback:", error);

        setDirectories(previousDirectories);
      }
    },
    [directories, setDirectories],
  );

  return {
    createDirectory,
    deleteDirectory,
    renameDirectory,
    moveDirectory,
  };
}

// src/features/studio/hooks/usePromptUpload.ts

import { useState } from "react";
import { api } from "@/src/services/api";
import { optimizeAndLogImage } from "@/src/utils/lib/image-optimizer";
import toast from "react-hot-toast";

export function usePromptUpload() {
  const [isUploading, setIsUploading] = useState(false);

  const uploadPrompt = async (fileOriginal: File) => {
    setIsUploading(true);
    console.log(
      `🚀 [UPLOAD-START] Iniciando flujo de subida para: ${fileOriginal.name} (${(fileOriginal.size / 1024 / 1024).toFixed(2)} MB)`,
    );

    try {
      // 🟢 PASO 1: Optimización y Transformación a WebP
      console.log(`⏳ [UPLOAD-STEP-1] Optimizando imagen en el cliente...`);
      const archivoOptimizado = await optimizeAndLogImage(fileOriginal);

      // 🟢 PASO 2: Pedir URL pre-firmada
      console.log(
        `⏳ [UPLOAD-STEP-2] Solicitando URL pre-firmada al Backend para: ${archivoOptimizado.name} (${archivoOptimizado.type})`,
      );
      const resUrl = await api.prompts.getPresignedUrl({
        filename: archivoOptimizado.name,
        contentType: archivoOptimizado.type,
      });

      if (!resUrl.ok) {
        throw new Error(`Error del servidor al obtener firma (Status: ${resUrl.status})`);
      }

      const { url, public_url } = await resUrl.json();
      console.log(`✅ [UPLOAD-STEP-2] Firma obtenida. URL Pública esperada: ${public_url}`);

      // 🟢 PASO 3: Subida directa a R2
      console.log(`☁️ [UPLOAD-STEP-3] Transfiriendo bytes a Cloudflare R2...`);
      const uploadRes = await fetch(url, {
        method: "PUT",
        body: archivoOptimizado,
        headers: {
          "Content-Type": archivoOptimizado.type,
          "Cache-Control": "public, max-age=31536000, immutable",
        },
      });

      if (!uploadRes.ok) {
        console.error(
          `❌ [UPLOAD-STEP-3] Cloudflare R2 rechazó el archivo. Status: ${uploadRes.status} - ${uploadRes.statusText}`,
        );
        throw new Error("Fallo al subir imagen al almacenamiento de Cloudflare");
      }

      console.log(`✅ [UPLOAD-STEP-3] Transferencia a R2 completada con éxito.`);

      // 🟢 PASO 4: Finalización
      console.log(
        `🎉 [UPLOAD-FINISH] Flujo de subida perfecto. Pasando el testigo a la Cuarentena del Backend.`,
      );

      return {
        success: true,
        image_url: public_url,
      };
    } catch (error: any) {
      console.error(`🚨 [UPLOAD-CRASH] El flujo de subida fue interrumpido:`, error);
      toast.error(error.message || "Fallo en la subida");
      return { success: false };
    } finally {
      setIsUploading(false);
    }
  };

  return { uploadPrompt, isUploading };
}

import { useState } from "react";
import { useSensor, useSensors, PointerSensor, DragStartEvent, DragEndEvent } from "@dnd-kit/core";
import { Directory } from "@/src/store/useStudioUI";
import { useDirectoryActions } from "./useDirectoryActions";

export function useStudioDnd() {
  const { moveDirectory } = useDirectoryActions();
  const [activeDragDir, setActiveDragDir] = useState<Directory | null>(null);

  const sensors = useSensors(
    useSensor(PointerSensor, {
      activationConstraint: { distance: 8 },
    }),
  );

  const handleDragStart = (event: DragStartEvent) => {
    const { active } = event;
    if (active.data.current?.type === "directory") {
      setActiveDragDir(active.data.current.directory);
    }
  };

  const handleDragEnd = (event: DragEndEvent) => {
    setActiveDragDir(null);
    const { active, over } = event;

    if (!over) return;

    const isTargetRoot = over.id === "root-zone";
    const targetDirId = isTargetRoot ? null : String(over.id).replace("dir-", "");
    const draggedDirId = String(active.id).replace("dir-", "");

    if (draggedDirId === (targetDirId || "root")) return;

    if (active.data.current?.type === "directory") {
      moveDirectory(draggedDirId, targetDirId);
    }
  };

  return { sensors, activeDragDir, handleDragStart, handleDragEnd };
}

import { useState, useEffect } from "react";
import { useStudioUI } from "@/src/store/useStudioUI";
import { useAuthStore } from "@/src/store/authStore";
import { fetchSeguro } from "@/src/services/api/client";
import toast from "react-hot-toast";

const API_URL = process.env.NEXT_PUBLIC_API_URL || "http://localhost:8000/api";

export function useStudioPrompts() {
  const { activeDirectoryId, searchQuery } = useStudioUI();
  const user = useAuthStore((state) => state.user);

  const [debouncedQuery, setDebouncedQuery] = useState(searchQuery);
  const [prompts, setPrompts] = useState<any[]>([]);
  const [isLoading, setIsLoading] = useState(false);

  useEffect(() => {
    const handler = setTimeout(() => setDebouncedQuery(searchQuery), 1000);
    return () => clearTimeout(handler);
  }, [searchQuery]);

  useEffect(() => {
    const fetchPrompts = async () => {
      setIsLoading(true);
      try {
        const API_BASE = process.env.NEXT_PUBLIC_API_URL || "http://localhost:8000/api";

        const endpoint = debouncedQuery
          ? `${API_BASE}/search/studio?q=${debouncedQuery}`
          : `${API_BASE}/prompts/studio?directory_id=${activeDirectoryId || ""}`;

        const res = await fetchSeguro(endpoint, { credentials: "include" });

        if (res.ok) {
          const data = await res.json();
          setPrompts(data.results || data.data || []);
        } else {
          console.error("Fallo del servidor:", res.status);
        }
      } catch (err) {
        console.error("Error en Studio Grid:", err);
      } finally {
        setIsLoading(false);
      }
    };

    fetchPrompts();
  }, [activeDirectoryId, debouncedQuery]);

  const deletePrompt = async (promptId: string) => {
    const toastId = toast.loading("Exterminando rastro digital...");
    try {
      const res = await fetchSeguro(`${API_URL}/prompts/${promptId}`, {
        method: "DELETE",
        credentials: "include",
      });

      if (res.ok) {
        setPrompts((prev) => prev.filter((p) => p.id !== promptId));
        toast.success("Eliminado del Studio", { id: toastId });
      } else {
        const errorData = await res.json();
        throw new Error(errorData.detail || "Fallo de autorización");
      }
    } catch (error: any) {
      console.error("Fallo al borrar:", error);
      toast.error(error.message || "Error al eliminar", { id: toastId });
    }
  };

  return {
    prompts,
    isLoading,
    user,
    deletePrompt,
  };
}

import { useState } from "react";
import { useRouter } from "next/navigation";
import { api } from "@/src/services/api";
import { useAuthStore } from "@/src/store/authStore";
import { slugifyUsername } from "@/src/utils/lib/string";
import toast from "react-hot-toast";

export function useUpdateUsername(lang: string) {
  const router = useRouter();
  const updateLocalUsername = useAuthStore((state) => state.updateLocalUsername);
  const [isLoading, setIsLoading] = useState(false);

  const handleUpdate = async (newNameRaw: string) => {
    const safeName = slugifyUsername(newNameRaw);

    setIsLoading(true);
    try {
      await api.user.updateUsername(safeName);

      updateLocalUsername(safeName);

      toast.success("¡Identidad actualizada!");

      router.push(`/${lang}/${safeName}/settings/profile`);
    } catch (error: any) {
      toast.error(error.message || "Error al actualizar el nombre");
    } finally {
      setIsLoading(false);
    }
  };

  return { handleUpdate, isLoading };
}
