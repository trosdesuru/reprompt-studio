"use client";

import { motion } from "framer-motion";
import { ReactNode } from "react";

interface FadeInProps {
  children: ReactNode;
  duration?: number;
  delay?: number;
  y?: number;
  className?: string;
}

export function FadeIn({ children, duration = 0.5, delay = 0, y = 10, className }: FadeInProps) {
  return (
    <motion.div
      className={className}
      initial={{ opacity: 0, y: y }}
      animate={{ opacity: 1, y: 0 }}
      transition={{
        duration: duration,
        delay: delay,
        ease: [0.21, 0.47, 0.32, 0.98],
      }}
    >
      {children}
    </motion.div>
  );
}

import Image from "next/image";
import { IconCamera, IconLoader2 } from "@tabler/icons-react";

export interface UserAvatarProps {
  username: string;
  avatarUrl?: string | null;
  size?: "xs" | "sm" | "md" | "lg" | "xl" | "2xl";
  className?: string;
  editable?: boolean;
  isUploading?: boolean;
  onClick?: () => void;
}

export function UserAvatar({
  username,
  avatarUrl,
  size = "md",
  className = "",
  editable = false,
  isUploading = false,
  onClick,
}: UserAvatarProps) {
  const sizeClasses = {
    xs: "w-6 h-6 text-xs",
    sm: "w-8 h-8 text-xs",
    md: "w-10 h-10 text-md",
    lg: "w-16 h-16 text-lg",
    xl: "w-24 h-24 text-3xl",
    "2xl": "w-20 h-20 md:w-28 md:h-28",
  };

  return (
    <div
      className={`
        ${sizeClasses[size]} 
        shrink-0 rounded-full flex items-center justify-center overflow-hidden
        bg-accent relative group
        ${editable && !isUploading ? "cursor-pointer hover:border-accent" : ""}
        ${className}
      `}
      aria-label={`Avatar de ${username}`}
      onClick={onClick}
    >
      {isUploading && (
        <div className="absolute inset-0 z-20 bg-background/80 backdrop-blur-sm flex items-center justify-center">
          <IconLoader2 className="animate-spin text-accent" size={24} />
        </div>
      )}

      {editable && !isUploading && (
        <div className="absolute inset-0 z-10 bg-black/50 opacity-0 group-hover:opacity-100 transition-opacity duration-200 flex flex-col items-center justify-center text-white">
          <IconCamera size={20} className="mb-0.5" />
          <span className="text-[9px] font-mono uppercase font-bold tracking-widest">Cambiar</span>
        </div>
      )}

      {avatarUrl ? (
        <Image
          src={avatarUrl}
          alt={`Imagen de perfil de ${username}`}
          fill
          sizes="(max-width: 768px) 100vw, 33vw"
          className="object-cover"
        />
      ) : (
        <div className="w-full h-full bg-popover flex items-center justify-center border-2 border-accent rounded-full">
          <span className="text-secondary font-black font-prompt uppercase tracking-widest opacity-80 group-hover:scale-110 transition-transform duration-300">
            {username ? username.substring(0, 2) : "??"}
          </span>
        </div>
      )}
    </div>
  );
}

interface BrandTextProps {
  text: string;
  brandName?: string;
  className?: string;
}

export function BrandText({
  text,
  brandName = "reprompt",
  className = "",
}: BrandTextProps) {
  if (!text) return null;
  const parts = text.split(new RegExp(`(${brandName})`, "gi"));

  return (
    <span className={className}>
      {parts.map((part, index) => {
        if (part.toLowerCase() === brandName.toLowerCase()) {
          return (
            <span
              key={index}
              className="font-black font-mono tracking-tighter bg-clip-text text-transparent bg-linear-to-r from-primary to-foreground"
            >
              {part}
            </span>
          );
        }
        return <span key={index}>{part}</span>;
      })}
    </span>
  );
}

"use client";

import React, { useState, useId } from "react";
import { IconEye, IconEyeOff } from "@tabler/icons-react";

export interface InputProps extends React.InputHTMLAttributes<HTMLInputElement> {
  label?: string;
  error?: string;
  icon?: React.ReactNode;
}

export const Input = React.forwardRef<HTMLInputElement, InputProps>(
  ({ label, error, icon, type = "text", className = "", ...props }, ref) => {
    const autoId = useId();
    const inputId = props.id || autoId;

    const [isPasswordVisible, setIsPasswordVisible] = useState(false);
    const isPasswordType = type === "password";
    const currentType = isPasswordType && isPasswordVisible ? "text" : type;

    return (
      <div className="space-y-2 w-full">
        {label && (
          <label
            htmlFor={inputId}
            className="text-xs font-black font-mono uppercase tracking-widest text-muted-foreground block"
          >
            {label}
          </label>
        )}

        <div className="relative">
          {icon && (
            <span className="absolute left-4 top-1/2 -translate-y-1/2 text-muted-foreground">
              {icon}
            </span>
          )}

          <input
            ref={ref}
            id={inputId}
            type={currentType}
            className={`
              w-full bg-brackground border-2 border-color rounded-lg p-2 font-mono text-foreground
              focus:outline-none focus:ring-2 focus:ring-accent-color transition-all
              overflow-auto
              ${icon ? "pl-12" : ""} 
              ${isPasswordType ? "pr-12" : ""}
              ${error ? "border-red-500 focus:ring-red-500" : ""}
              ${className}
            `}
            {...props}
          />

          {isPasswordType && (
            <button
              type="button"
              onClick={() => setIsPasswordVisible(!isPasswordVisible)}
              className="absolute right-3 top-1/2 -translate-y-1/2 p-1 text-muted-foreground hover:text-foreground transition-colors"
              aria-label={isPasswordVisible ? "Ocultar contraseña" : "Mostrar contraseña"}
              tabIndex={-1}
            >
              {isPasswordVisible ? <IconEye size={20} /> : <IconEyeOff size={20} />}
            </button>
          )}
        </div>

        {error && <p className="text-xs font-bold text-red-500 animate-fade-in">{error}</p>}
      </div>
    );
  },
);

Input.displayName = "Input";

const TooltipProvider = TooltipPrimitive.Provider;

const TooltipContent = React.forwardRef<
  React.ElementRef<typeof TooltipPrimitive.Content>,
  React.ComponentPropsWithoutRef<typeof TooltipPrimitive.Content>
>(({ className, sideOffset = 4, ...props }, ref) => (
  <TooltipPrimitive.Portal>
    <TooltipPrimitive.Content
      ref={ref}
      sideOffset={sideOffset}
      className={cn(
        "z-100 overflow-hidden rounded-md shadow-lg animate-in fade-in-0 zoom-in-95 data-[state=closed]:animate-out data-[state=closed]:fade-out-0 data-[state=closed]:zoom-out-95 data-[side=bottom]:slide-in-from-top-2 data-[side=left]:slide-in-from-right-2 data-[side=right]:slide-in-from-left-2 data-[side=top]:slide-in-from-bottom-2",
        className,
      )}
      {...props}
    />
  </TooltipPrimitive.Portal>
));
TooltipContent.displayName = TooltipPrimitive.Content.displayName;

interface SimpleTooltipProps {
  content: React.ReactNode;
  children: React.ReactNode;
  side?: "top" | "right" | "bottom" | "left";
  delayDuration?: number;
  variant?: "default" | "rich";
  className?: string;
}

export function Tooltip({
  content,
  children,
  className,
  side = "top",
  delayDuration = 200,
  variant = "default",
}: SimpleTooltipProps) {
  const styles = {
    default: {
      container:
        "bg-foreground text-background px-3 py-1.5 text-[10px] font-mono font-bold uppercase tracking-widest",
      arrow: "fill-foreground",
    },
    rich: {
      container: "bg-surface text-popover-foreground border border-border px-3 py-2 neo-shadow-sm",
      arrow: "fill-popover",
    },
  };

  return (
    <TooltipProvider delayDuration={delayDuration}>
      <TooltipRoot>
        <TooltipTrigger asChild>{children}</TooltipTrigger>
        <TooltipContent side={side} className={cn(styles[variant].container, className)}>
          {content}
          <TooltipPrimitive.Arrow className={styles[variant].arrow} />
        </TooltipContent>
      </TooltipRoot>
    </TooltipProvider>
  );
}

            <span className="text-2xl font-black text-foreground">{currentMb.toFixed(2)}</span>
            <span className="text-xs font-mono text-muted-foreground mb-1">
              / {limitMb.toLocaleString()} MB
            </span>
          </div>
        </div>
        <span className="text-xs font-mono font-bold">{percentage.toFixed(1)}%</span>

        {...props}
      />

        type="file"
        ref={fileInputRef}
        className="hidden"
        accept="image/png, image/jpeg, image/webp"
        onChange={handleFileChange}
      />
    </>
  );

interface ModalProps {
  isOpen: boolean;
  onClose: () => void;
  children: React.ReactNode;
  maxWidth?: string;
}

export function Modal({ isOpen, onClose, children, maxWidth = "max-w-md" }: ModalProps) {
  const [mounted, setMounted] = useState(false);

  useEffect(() => {
    setMounted(true);
  }, []);

  useEffect(() => {
    const handleEsc = (e: KeyboardEvent) => {
      if (e.key === "Escape") onClose();
    };
    if (isOpen) {
      window.addEventListener("keydown", handleEsc);
    }
    return () => window.removeEventListener("keydown", handleEsc);
  }, [isOpen, onClose]);

  if (!isOpen || !mounted) return null;

  return createPortal(
    <div className="fixed inset-0 z-9999 flex items-center justify-center p-4 bg-background/80 backdrop-blur-sm animate-in fade-in duration-200">
      <div
        className={`
          bg-background border border-border/50 rounded-xl w-full ${maxWidth} max-h-[90dvh] overflow-y-auto animate-in zoom-in-95 duration-200 relative flex flex-col
          shadow-[0_0_40px_rgba(0,0,0,0.1)] dark:shadow-[0_0_40px_rgba(255,255,255,0.06)]
          `}
      >
        <button
          onClick={onClose}
          className="absolute top-4 right-4 p-1 text-muted-foreground hover:text-foreground transition-colors z-10 rounded-md focus:outline-none focus:ring-2 focus:ring-primary/50"
          aria-label="Cerrar modal"

  const [isModalOpen, setIsModalOpen] = useState(false);
  const { query, setQuery, handleSearch, isLoading } = useSearch();

  const modalContentRef = useRef<HTMLDivElement>(null);

  useClickOutside(modalContentRef, () => {
    if (isModalOpen) handleCloseAndClear();

              query={query}
              isLoading={isLoading}
              placeholder="Redefine tu búsqueda"
              onQueryChange={setQuery}
              onSubmit={onExecuteSearch}
              onClear={() => setQuery("")}
              enableKeyboardShortcut={true}
            />

          </div>

  label: string;
}

interface Props {
  options: Option[];
  activeCategories: Record<string, boolean>;
  onToggle: (id: string) => void;
}

export function SearchCategorySelector({ options, activeCategories, onToggle }: Props) {
  const anchor = useComboboxAnchor();

    options.filter(c => activeCategories[c.id]).map(c => c.label)
  , [options, activeCategories]);

      const isActive = !!activeCategories[c.id];

    });

      multiple
      items={options.map(c => c.label)}
      value={selectedLabels}
      onValueChange={handleValueChange}

        ref={anchor}

"use client";

import { useRef, useEffect } from "react";
import { IconSearch, IconLoader2, IconX } from "@tabler/icons-react";
import { Tooltip } from "../../atoms/Tooltip/Tooltip";

interface UniversalSearchBarProps {
  query: string;
  isLoading: boolean;
  placeholder?: string;
  onQueryChange: (q: string) => void;
  onSubmit: () => void;
  onClear: () => void;
  enableKeyboardShortcut?: boolean;
}

export function UniversalSearchBar({
  query,
  isLoading,
  placeholder = "Buscar...",
  onQueryChange,
  onSubmit,
  onClear,
  enableKeyboardShortcut = false,
}: UniversalSearchBarProps) {
  const inputRef = useRef<HTMLInputElement>(null);

  useEffect(() => {
    if (!enableKeyboardShortcut) return;
    const handleKeyDown = (e: KeyboardEvent) => {
      const target = e.target as HTMLElement;
      const isInput =
        target.tagName === "INPUT" || target.tagName === "TEXTAREA" || target.isContentEditable;
      if (e.key === "/" && !isInput) {
        e.preventDefault();
        inputRef.current?.focus();
      }
    };
    window.addEventListener("keydown", handleKeyDown);
    return () => window.removeEventListener("keydown", handleKeyDown);
  }, [enableKeyboardShortcut]);

  useEffect(() => {
    const timer = setTimeout(() => {
      inputRef.current?.focus();
    }, 50);
    return () => clearTimeout(timer);
  }, []);

  return (
    <div className="relative flex items-center w-full h-10 bg-muted/20 border border-border/40 rounded-xl focus-within:border-(--accent-color)/50 focus-within:ring-1 focus-within:ring-(--accent-color)/20 transition-all backdrop-blur-sm">
      <form
        onSubmit={(e) => {
          e.preventDefault();
          onSubmit();
        }}
        className="flex-1 h-full flex items-center relative pl-3"
      >
        <div className="flex items-center justify-center w-5 h-5">
          {isLoading ? (
            <IconLoader2 size={16} className="animate-spin text-(--accent-color)" />
          ) : (
            <IconSearch size={16} className="text-muted-foreground/60" />
          )}
        </div>

        <input
          ref={inputRef}
          value={query}
          onChange={(e) => onQueryChange(e.target.value)}
          className="w-full h-full bg-transparent px-3 text-sm outline-none placeholder:text-muted-foreground/40 font-medium"
          placeholder={placeholder}
        />

        {query && (
          <button
            type="button"
            onClick={onClear}
            className="pr-3 text-muted-foreground/40 hover:text-foreground transition-colors"
          >
            <Tooltip className="z-9999" content="Cerrar">
              <IconX size={14} />
            </Tooltip>
          </button>
        )}
      </form>
    </div>
  );
}

"use client";

import { useState } from "react";
import { useStudioUI, Directory } from "@/src/store/useStudioUI";
import { SidebarDirectoryItem } from "../../atoms/Studio/SidebarDirectoryItem";
import { useDirectoryDnd } from "@/src/features/studio/hooks/useDirectoryDnd";
import { useDirectoryActions } from "@/src/features/studio/hooks/useDirectoryActions";

interface Props {
  directory: Directory;
  level?: number;
}

export function SidebarDirectory({ directory, level = 0 }: Props) {
  const [isOpen, setIsOpen] = useState(false);
  const { getChildDirectories, activeDirectoryId, setActiveDirectory } = useStudioUI();

  const { createDirectory, deleteDirectory, renameDirectory } = useDirectoryActions();

  const { setBothRefs, isDragging, isOver, attributes, listeners } = useDirectoryDnd(directory);

  const children = getChildDirectories(directory.id);
  const hasChildren = children.length > 0;
  const isSelected = activeDirectoryId === directory.id;

  const handleCreateSubdirectory = async (onSuccess: () => void) => {
    const name = window.prompt("Nombre de la sub-carpeta:");
    if (name?.trim()) {
      await createDirectory(name.trim(), directory.id);
      onSuccess();
    }
  };

  if (isDragging) return null;

  return (
    <div className="flex flex-col w-full" ref={setBothRefs}>
      <SidebarDirectoryItem
        directory={directory}
        level={level}
        isOpen={isOpen}
        hasChildren={hasChildren}
        isSelected={isSelected}
        isOver={isOver}
        isDragging={isDragging}
        setBothRefs={setBothRefs}
        attributes={attributes}
        listeners={listeners}
        isEffectivelyPrivate={directory.is_private || false}
        onToggle={() => setIsOpen(!isOpen)}
        onSelect={() => setActiveDirectory(directory.id)}
        onCreateSub={(e) => {
          e.stopPropagation();
          handleCreateSubdirectory(() => setIsOpen(true));
        }}
        onDelete={(e) => {
          e.stopPropagation();
          if (window.confirm(`¿Borrar "${directory.name}"?`)) {
            deleteDirectory(directory.id);
          }
        }}
        onRename={(e) => {
          e.stopPropagation();
          const newName = window.prompt("Nuevo nombre:", directory.name);
          if (newName && newName.trim() !== directory.name) {
            renameDirectory(directory.id, newName.trim());
          }
        }}
      />

      {isOpen && hasChildren && (
        <div className="flex flex-col w-full">
          {children.map((child: Directory) => (
            <SidebarDirectory key={child.id} directory={child} level={level + 1} />
          ))}
        </div>
      )}
    </div>
  );
}

"use client";

import { useEffect, useState } from "react";
import { api } from "@/src/services/api";
import { StorageStatsResponse } from "@/src/services/api/admin";
import toast from "react-hot-toast";
import {
  IconCloudDataConnection,
  IconFiles,
  IconAlertCircle,
  IconServer,
  IconFlame,
} from "@tabler/icons-react";
import { QuotaBar } from "../../molecules/Admin/QuotaBar";

export function StorageModule() {
  const [stats, setStats] = useState<StorageStatsResponse | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const [isPurging, setIsPurging] = useState(false);
  const [confirmPurge, setConfirmPurge] = useState(false);

  const fetchStorage = async () => {
    try {
      const data = await api.admin.getStorageStats();
      setStats(data);
    } catch (err: any) {
      setError(err.message || "Error al conectar con Cloudflare R2");
    }
  };

  useEffect(() => {
    let isMounted = true;

    const initialFetch = async () => {
      try {
        setIsLoading(true);
        const data = await api.admin.getStorageStats();
        if (isMounted) setStats(data);
      } catch (err: any) {
        if (isMounted) setError(err.message || "Error al conectar con Cloudflare R2");
      } finally {
        if (isMounted) setIsLoading(false);
      }
    };

    initialFetch();
    return () => {
      isMounted = false;
    };
  }, []);

  const handlePurge = async () => {
    if (!confirmPurge) {
      setConfirmPurge(true);
      setTimeout(() => setConfirmPurge(false), 4000);
      return;
    }

    const finalConfirm = prompt(
      "ESTO BORRARÁ TODO EL BUCKET R2.\nEscribe DESTRUIR para confirmar:",
    );
    if (finalConfirm !== "DESTRUIR") {
      setConfirmPurge(false);
      return;
    }

    setIsPurging(true);
    try {
      toast.loading("Aniquilando bucket R2...", { id: "purge" });
      const result = await api.admin.purgeStorage();
      toast.success(result.message, { id: "purge", duration: 5000 });
      await fetchStorage();
    } catch (error: any) {
      toast.error(error.message || "Fallo al purgar R2", { id: "purge" });
    } finally {
      setIsPurging(false);
      setConfirmPurge(false);
    }
  };

  if (error) {
    return (
      <div className="p-6 bg-red-500/10 border border-red-500/20 rounded-xl flex items-start gap-3 text-red-500">
        <IconAlertCircle className="shrink-0 mt-0.5" />
        <div>
          <h3 className="font-bold">Alerta de Infraestructura (R2)</h3>
          <p className="text-sm opacity-80">{error}</p>
        </div>
      </div>
    );
  }

  if (isLoading || !stats) {
    return <StorageModuleSkeleton />;
  }

  const isStorageHealthy = stats.total_mb < 10000;

  return (
    <div className="space-y-6 animate-in fade-in duration-500">
      <QuotaBar
        label="Cuota Mensual Cloudflare R2 (Free Tier)"
        currentMb={stats.total_mb}
        limitMb={stats.limit_mb}
      />

      <div
        className={`p-4 border rounded-xl flex items-center justify-between shadow-sm ${
          isStorageHealthy
            ? "bg-green-500/10 border-green-500/20"
            : "bg-yellow-500/10 border-yellow-500/20"
        }`}
      >
        <div className="flex items-center gap-3">
          <IconCloudDataConnection
            className={isStorageHealthy ? "text-green-500" : "text-yellow-500"}
          />
          <div>
            <h3
              className={`font-bold text-sm uppercase tracking-widest ${isStorageHealthy ? "text-green-600 dark:text-green-400" : "text-yellow-600 dark:text-yellow-400"}`}
            >
              Cloudflare R2 (Object Storage)
            </h3>
            <p className="text-xs text-muted-foreground font-mono">
              {isStorageHealthy ? "Operativo y Saludable" : "Acercándose a cuota límite"}
            </p>
          </div>
        </div>
        <div className="flex items-center gap-2 text-xs font-mono">
          <span className="relative flex h-3 w-3">
            <span
              className={`animate-ping absolute inline-flex h-full w-full rounded-full opacity-75 ${isStorageHealthy ? "bg-green-400" : "bg-yellow-400"}`}
            ></span>
            <span
              className={`relative inline-flex rounded-full h-3 w-3 ${isStorageHealthy ? "bg-green-500" : "bg-yellow-500"}`}

              Ancho de Disco
            </p>

              {Number(stats.total_mb).toFixed(2)}{" "}
              <span className="text-sm text-muted-foreground font-normal">MB</span>
            </p>
          </div>
          <div className="p-3 bg-muted/30 rounded-lg text-blue-500">

"use client";

import { useCallback } from "react";
import { useDropzone } from "react-dropzone";
import { DropzoneLoading } from "./DropzoneLoading";
import { DropzoneContent } from "./DropzoneContent";
import { DropzoneError } from "./DropzoneError";

interface ExtractorDropzoneProps {
  onFileDrop: (acceptedFiles: File[], rejectedFiles: any[]) => void;
  isExtracting: boolean;
  error: string | null;
}

export function ExtractorDropzone({
  onFileDrop,
  isExtracting,
  error: externalError,
}: ExtractorDropzoneProps) {
  const onDrop = useCallback(
    (acceptedFiles: File[], fileRejections: any[]) => {
      if (acceptedFiles.length > 0 || fileRejections.length > 0) {
        onFileDrop(acceptedFiles, fileRejections);
      }
    },
    [onFileDrop],
  );

  // 🟢 Extraemos isDragReject
  const { getRootProps, getInputProps, isDragActive, isDragReject } = useDropzone({
    onDrop,
    accept: {
      "image/png": [".png"],
      "image/jpeg": [".jpg", ".jpeg"],
      "image/webp": [".webp"], // Sugiero añadir webp
      "text/plain": [".txt"],
      "application/json": [".json"],
      "application/zip": [".zip"],
      "application/x-zip-compressed": [".zip"],
    },
    disabled: isExtracting,
  });

  if (isExtracting) return <DropzoneLoading />;

  return (
    <div
      {...getRootProps()}
      className={`group relative w-full border-4 border-dashed transition-all duration-300 ease-in-out flex flex-col items-center justify-center min-h-400px cursor-pointer neo-border overflow-hidden
        ${
          isDragReject
            ? "border-destructive bg-destructive/5"
            : isDragActive
              ? "border-(--accent-color) bg-(--accent-color)/5 scale-[1.01] neo-shadow-accent"
              : "border-border bg-muted/5 hover:bg-muted/10 hover:border-foreground/30"
        }`}
    >
      <input {...getInputProps()} />
      <DropzoneContent isDragActive={isDragActive} isDragReject={isDragReject} />

      {externalError && <DropzoneError message={externalError} />}
    </div>
  );
}

import { BatchItem, SkippedFile } from "@/src/features/extractor/hooks/usePromptExtractor";

import { handleError } from "@/src/utils/errors/ErrorHandler";
import { useNotificationStore } from "@/src/store/notificationStore";
import { fetchSeguro } from "@/src/services/api/client";
import { optimizeAndLogImage } from "@/src/utils/lib/image-optimizer";

interface Props {
  batch: BatchItem[];
  skippedFiles: SkippedFile[];
  onUpdateStatus: (id: string, status: BatchItem["status"]) => void;
  onUpdateItemData: (id: string, newData: Partial<ExtractedPromptData>) => void;
  onRemoveItem: (id: string) => void;
  onClearAll: () => void;
  onClearSkipped: () => void;
}

export function BatchInspector({
  batch,
  skippedFiles,
  onUpdateStatus,
  onUpdateItemData,
  onRemoveItem,
  onClearAll,
  onClearSkipped,
}: Props) {
  const [isGlobalLoading, setIsGlobalLoading] = useState(false);
  const notify = useNotificationStore((state) => state.notify);

  const handleSaveItem = async (item: BatchItem) => {
    if (item.status === "success") return;
    onUpdateStatus(item.id, "saving");

    try {
      let finalWidth = 0;
      let finalHeight = 0;
      let finalImageUrl: string | null = null; // 🟢 FIX: Declaramos la variable que faltaba

      if (item.previewUrl && item.previewUrl.startsWith("blob:")) {
        // 🟢 FIX: TypeScript ahora sabe que esto devuelve { file, width, height }
        const optimizedData = await optimizeAndLogImage(item.file);
        const archivoOptimizado = optimizedData.file;
        finalWidth = optimizedData.width;
        finalHeight = optimizedData.height;

        const API_URL = process.env.NEXT_PUBLIC_API_URL || "http://localhost:8000";

        const ticketRes = await fetchSeguro(`${API_URL}/prompts/generate-upload-url`, {
          method: "POST",
          credentials: "include",
          body: JSON.stringify({
            file_name: archivoOptimizado.name,
            content_type: archivoOptimizado.type,
          }),
        });

        if (!ticketRes.ok) throw new Error("No se pudo obtener el ticket de subida de R2");

        const ticketData = await ticketRes.json();
        const uploadUrl = ticketData.upload_url;

        const fetchOptions = {
          method: "PUT",
          headers: {
            "Content-Type": archivoOptimizado.type,
            "Cache-Control": "public, max-age=31536000, immutable",
          },
          body: archivoOptimizado,
        };

        const uploadRes = await fetch(uploadUrl, fetchOptions);

        if (!uploadRes.ok) {
          throw new Error("Fallo la subida física a Cloudflare R2");
        }

        finalImageUrl = ticketData.file_url;
      }

      const payload = {
        title: `${item.data.software || "IA"}`,
        prompt: item.data.prompt,
        negative_prompt: item.data.negativePrompt,
        software: item.data.software,
        model: item.data.model,
        sampler: item.data.sampler,
        steps: item.data.steps,
        cfg_scale: item.data.cfgScale,
        seed: item.data.seed,
        resolution: item.data.resolution,
        width: finalWidth,
        height: finalHeight,
        image_url: finalImageUrl,
        extra_data: {
          loras: item.data.loras || [],
          triggerWords: item.data.triggerWords || [],
        },
      };

      await api.savePrompt(payload);
      onUpdateStatus(item.id, "success");

      notify("Studio", "Prompt guardado correctamente", "success");

      setTimeout(() => {
        onRemoveItem(item.id);
      }, 1000);
    } catch (err: unknown) {
      handleError(err, "SaveBatchItem");
      onUpdateStatus(item.id, "error");
    }
  };

  const handleSaveAll = async () => {
    setIsGlobalLoading(true);

    await Promise.allSettled(pendingItems.map((item) => handleSaveItem(item)));
    setIsGlobalLoading(false);

  return (
    <div className="space-y-6 animate-fade-in">
      {/* FEEDBACK DE ARCHIVOS DESCARTADOS INTACTO AQUÍ */}

            className="absolute top-2 right-2 text-orange-700 hover:text-orange-900"
          >
            <IconX size={20} />
          </button>
          <div className="flex items-center gap-2 text-orange-600 font-black font-mono uppercase mb-2">
            <IconAlertTriangle size={20} /> Se descartaron {skippedFiles.length} archivos
          </div>

                    {f.name}

          <Button
            size="md"
            onClick={onClearAll}
            variant="tertiary"
            disabled={isGlobalLoading}
            className="text-sm!"
          >
            Descartar Todos

            variant="primary"
            onClick={handleSaveAll}
            isLoading={isGlobalLoading}
            className="text-sm! w-full sm:w-auto"

            <IconDeviceFloppy size={18} className="mr-2" /> Guardar Todo

import { usePathname, useSearchParams } from "next/navigation";
import { useProgressStore } from "@/src/store/progressStore";

export function GlobalProgress() {
  const { isAnimating, progress, start, done } = useProgressStore();
  const pathname = usePathname();
  const searchParams = useSearchParams();

  // 1. Cerrar barra al terminar la navegación de Next.js
  useEffect(() => {
    done();
  }, [pathname, searchParams, done]);

  // 2. Interceptor de enlaces nativos
  useEffect(() => {
    const handleAnchorClick = (e: MouseEvent) => {
      const anchor = (e.target as HTMLElement).closest("a");
      if (!anchor) return;

      const href = anchor.getAttribute("href");
      const target = anchor.getAttribute("target");

      if (href && href.startsWith("/") && target !== "_blank" && href !== pathname) {
        start();
      }
    };

    document.addEventListener("click", handleAnchorClick, true);
    return () => document.removeEventListener("click", handleAnchorClick, true);
  }, [pathname, start]);

  // 🛡️ FIX ARQUITECTÓNICO: NUNCA DEVOLVEMOS NULL.
  // Dejamos el elemento en el DOM para que el motor CSS pueda calcular las transiciones de salida (Fade out).

  return (
    <div className="fixed top-0 left-0 w-full z-99999 pointer-events-none">
      <div
        className="relative transition-all ease-out"
        style={{
          height: "3px", // El grosor elegante de la industria
          width: `${progress}%`,
          opacity: isAnimating ? 1 : 0,

import { useState, useEffect } from "react";

interface PageHeaderProps {
  title: string;
  subtitle?: string;
  description: string;
  icon: React.ReactNode;
  breadcrumbItems: any[];
  scrollContainerId?: string;
  collapseOnHide?: boolean;
}

export function PageHeader({
  title,
  subtitle,
  description,
  icon,
  breadcrumbItems,
  scrollContainerId,
  collapseOnHide = false,
}: PageHeaderProps) {
  const [isVisible, setIsVisible] = useState(true);
  const [lastScrollY, setLastScrollY] = useState(0);

  useEffect(() => {
    const target = scrollContainerId ? document.getElementById(scrollContainerId) : window;
    if (!target) return;

    const handleScroll = () => {
      const currentScrollY = scrollContainerId ? (target as HTMLElement).scrollTop : window.scrollY;

      if (currentScrollY > lastScrollY && currentScrollY > 100) {
        setIsVisible(false);
      } else if (currentScrollY < lastScrollY) {
        setIsVisible(true);
      }

      setLastScrollY(currentScrollY);
    };

    target.addEventListener("scroll", handleScroll, { passive: true });
    return () => target.removeEventListener("scroll", handleScroll);
  }, [lastScrollY, scrollContainerId]);

  const visibilityClasses = isVisible
    ? "opacity-100 translate-y-0 max-h-[300px] mb-4"
    : collapseOnHide
      ? "opacity-0 -translate-y-4 max-h-0 overflow-hidden mb-0"
      : "opacity-0 -translate-y-full";

  return (
    <div className={`transition-all duration-500 ease-in-out ${visibilityClasses}`}>
      <header
        className={`
          md:block w-full bg-background/80
          ${!collapseOnHide ? "sticky top-0 z-40" : "relative z-40"}
        `}
      >
        <div className="w-full mx-auto space-y-6">
          <div className="flex justify-between items-center gap-4">
            <div>
              <h1 className="text-xl sm:text-2xl text-foreground/90 font-sans font-black tracking-tighter flex items-center gap-2">

"use client";

import { useEffect, useRef, useCallback } from "react";
import { IconX, IconLoader2 } from "@tabler/icons-react";
import { UniversalMediaCard } from "@/src/system/organisms/Media/UniversalMediaCard";
import { mapViewerToMediaCard } from "@/src/utils/adapters/mediaAdapters";
import { PageLoader } from "@/src/system/atoms/Loader/PageLoader";

interface ViewerPage {
  image_url: string;
  name: string;
  prompt?: string;
}

interface ImageViewerModalProps {
  title: string;
  pages: ViewerPage[];
  isLoading: boolean;
  isLoadingMore: boolean;
  hasMorePages: boolean;
  onLoadMore: () => void;
  onClose: () => void;
}

export function ImageViewerModal({
  title,
  pages,
  isLoading,
  isLoadingMore,
  hasMorePages,
  onLoadMore,
  onClose,
}: ImageViewerModalProps) {
  const observerRef = useRef<IntersectionObserver | null>(null);

  const lastElementRef = useCallback(
    (node: HTMLDivElement | null) => {
      if (isLoading || isLoadingMore) return;
      if (observerRef.current) observerRef.current.disconnect();

      observerRef.current = new IntersectionObserver((entries) => {
        if (entries[0].isIntersecting && hasMorePages) onLoadMore();
      });

      if (node) observerRef.current.observe(node);
    },
    [isLoading, isLoadingMore, hasMorePages, onLoadMore],
  );

  useEffect(() => {
    const handleEsc = (e: KeyboardEvent) => {
      if (e.key === "Escape") onClose();
    };
    window.addEventListener("keydown", handleEsc);
    document.body.style.overflow = "hidden";
    return () => {
      window.removeEventListener("keydown", handleEsc);
      document.body.style.overflow = "unset";
    };
  }, [onClose]);

  return (
    <div className="fixed inset-0 z-100 flex flex-col bg-background/95 backdrop-blur-xl animate-in fade-in duration-200">

        <h2 className="text-lg md:text-xl font-bold line-clamp-1">{title}</h2>
        <button
          onClick={onClose}
          className="p-2 rounded-full hover:bg-muted text-muted-foreground hover:text-foreground transition-colors"
        >
          <IconX size={24} />
        </button>
      </div>

      <div className="flex-1 overflow-y-auto custom-scrollbar p-4 md:p-8">
        {isLoading && pages.length === 0 ? (
          <div className="w-full h-full flex flex-col items-center justify-center gap-4 text-muted-foreground">
            <PageLoader text="Cargando Imagen ..."/>
          </div>
        ) : (
          <div className="max-w-7xl mx-auto space-y-12">
            {pages.map((page, index) => (
              <div
                key={`${index}-${page.image_url}`}
                ref={index === pages.length - 1 ? lastElementRef : null}
                className="w-full flex justify-center"
              >
                <div className="w-full max-w-4xl">
                  <UniversalMediaCard data={mapViewerToMediaCard(page, index)} variant="viewer" />
                </div>
              </div>
            ))}
            {isLoadingMore && (
              <div className="py-8 flex justify-center">
                <IconLoader2 size={32} className="animate-spin text-muted-foreground" />
              </div>
            )}
          </div>
        )}
      </div>
    </div>
  );
}

"use client";

import { useMemo } from "react";
import { useAuthStore } from "@/src/store/authStore";
import { FeedItem } from "@/src/types/models";
import { FeedHeader } from "@/src/system/molecules/Feed/FeedHeader";
import { FeedSentinel } from "@/src/system/molecules/Feed/FeedSentinel";
import { PublicFeedGrid } from "./PublicFeedGrid";
import { useInfiniteScroll } from "@/src/utils/hooks/useInfiniteScroll";

interface FeedLayoutProps {
  title?: string;
  feed: FeedItem[];
  isLoading: boolean;
  lang: string;
  hasMore: boolean;
  onLoadMore: () => void;
  onFilterClick?: () => void;
}

export function FeedLayout({
  title = "Explora la Comunidad",
  feed,
  isLoading,
  lang,
  hasMore,
  onLoadMore,
  onFilterClick,
}: FeedLayoutProps) {
  const { user } = useAuthStore();

  const filteredFeed = useMemo(() => {
    if (!user || !feed) return feed || [];

    return feed.filter((item) => {
      const itemAuthor = item.username;

      return itemAuthor !== user.username;
    });
  }, [feed, user]);

  const sentinelRef = useInfiniteScroll({ hasMore, isLoading, onLoadMore });

  return (
    <div className="pt-8 w-full">
      <FeedHeader title={title} count={filteredFeed.length} onFilterClick={onFilterClick} />

      <PublicFeedGrid feed={filteredFeed} lang={lang} />

      <FeedSentinel
        ref={sentinelRef}
        isLoading={isLoading}
        hasMore={hasMore}
        isEmpty={filteredFeed.length === 0}
      />
    </div>
  );
}

"use client";

import { useMemo } from "react";
import { useParams } from "next/navigation";
import { PageLoader } from "@/src/system/atoms/Loader/PageLoader";
import { DirectoryHeader } from "@/src/system/molecules/Studio/DirectoryHeader";
import { PromptGrid } from "@/src/system/organisms/Studio/PromptGrid";

import { useStudioPrompts } from "@/src/features/studio/hooks/useStudioPrompts";
import { useStudioUI } from "@/src/store/useStudioUI";
import { useStudioSync } from "@/src/features/studio/hooks/useStudioSync";

export function StudioWorkspace() {
  useStudioSync();

  const params = useParams();
  const lang = (params?.lang as string) || "es";

  const { prompts, isLoading, user, deletePrompt } = useStudioPrompts();
  const { activeDirectoryId, directories } = useStudioUI();

  const activeDir = useMemo(
    () => directories.find((d) => String(d.id) === String(activeDirectoryId)),
    [activeDirectoryId, directories],
  );

  const directoryName = activeDirectoryId && activeDir ? activeDir.name : "reprompt root";
  const authorUsername = user?.username || "unknown";

  if (isLoading) {
    return (
      <div className="h-full flex flex-col flex-1 items-center justify-center">
        <PageLoader text="Sincronizando Studio..." />
      </div>
    );
  }

  return (
    <>
      <DirectoryHeader name={directoryName} />
      <PromptGrid
        prompts={prompts}
        lang={lang}
        authorUsername={authorUsername}
        onDelete={deletePrompt}
      />
    </>
  );
}
