def run_network_diagnostics():
    logger.info("============== 🕵️ INICIANDO DIAGNÓSTICO DE RED ==============")
    parsed = urlparse(clean_url)
    host = parsed.hostname
    port = parsed.port or 5432

    # 1. Prueba de DNS
    logger.info(f"📡 [PROBE 1/3] Resolviendo DNS para: {host}")
    try:
        ip = socket.gethostbyname(host)
        logger.info(f"✅ [PROBE 1/3] IP Resuelta: {ip}")
    except Exception as e:
        logger.error(f"❌ [PROBE 1/3] Fallo de resolución DNS: {e}")
        return False

    # 2. Prueba de Socket TCP (Ver si el puerto 5432 está abierto y accesible)
    logger.info(f"🚪 [PROBE 2/3] Probando puerto TCP {ip}:{port}...")
    try:
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.settimeout(5.0)  # 5 segundos de timeout
        sock.connect((ip, port))
        logger.info("✅ [PROBE 2/3] Puerto TCP abierto y accesible.")
        sock.close()
    except Exception as e:
        logger.error(f"❌ [PROBE 2/3] El puerto está bloqueado o inaccesible: {e}")
        return False

    # 3. Prueba de Autenticación Cruda y SSL (psycopg2 directo, sin SQLAlchemy)
    logger.info("🔐 [PROBE 3/3] Probando Handshake SSL y Autenticación directa...")
    try:
        start_time = time.time()
        # Usamos un timeout corto de 10s para no esperar los famosos 19.5s
        conn = psycopg2.connect(clean_url, connect_timeout=10)
        elapsed = round(time.time() - start_time, 2)
        logger.info(f"✅ [PROBE 3/3] ¡Autenticación SSL exitosa en {elapsed}s!")

        # Test de lectura básica
        cur = conn.cursor()
        cur.execute("SELECT version();")
        version = cur.fetchone()
        logger.info(f"🐘 [PROBE] Versión de DB: {version[0][:30]}...")
        cur.close()
        conn.close()
        logger.info("============== 🕵️ DIAGNÓSTICO COMPLETADO ==============\n")
        return True
    except Exception as e:
        elapsed = round(time.time() - start_time, 2)
        logger.error(f"❌ [PROBE 3/3] Falló la negociación SSL/Auth tras {elapsed}s.")
        logger.error(f"Detalle exacto: {e}")
        logger.info("============== 🕵️ DIAGNÓSTICO COMPLETADO ==============\n")
        return False


def init_db():
    logger.info("🔌 [DB] Verificando infraestructura antes de iniciar...")

    # 🟢 EJECUTAMOS EL DIAGNÓSTICO ANTES DE NADA
    run_network_diagnostics()

    max_retries = 3
    retry_delay = 5

    logger.info(
        "🔌 [DB] Iniciando secuencia de sincronización de tablas (SQLAlchemy)..."
    )
    connection_successful = False

    for attempt in range(max_retries):
        try:
            logger.info(
                f"⏳ [DB] [Intento {attempt + 1}/{max_retries}] Sincronizando metadata..."
            )
            SQLModel.metadata.create_all(engine)
            logger.info("✅ [DB] ¡Metadata sincronizada correctamente!")
            connection_successful = True
            break
        except Exception as e:
            logger.warning(f"⚠️ [DB] Falló el intento {attempt + 1}. Detalle: {e}")
            if attempt < max_retries - 1:
                time.sleep(retry_delay)
            else:
                logger.error("❌ [DB] Imposible sincronizar tablas.")
                raise e

    if connection_successful:
        with Session(engine) as session:
            existing_code = session.exec(select(InviteCode)).first()
            if not existing_code:
                master_code = f"TROYA-{secrets.token_hex(4).upper()}"
                new_invite = InviteCode(code=master_code)
                session.add(new_invite)
                session.commit()
                logger.info("✨ Base de Datos inicializada por primera vez.")
                logger.warning(f"🔑 TU CÓDIGO MAESTRO ES: {master_code}")
            else:
                logger.info("✅ Base de datos lista para operar.")


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO)
    init_db()

# backend/app/api/deps.py
import os
import redis
import logging
import jwt
from fastapi import Depends, HTTPException, status, Request
from jwt.exceptions import InvalidTokenError, PyJWTError
from sqlmodel import Session, select
from app.core.redis_config import conn as redis_client
from app.core.db import get_session
from app.models.user import User

logger = logging.getLogger("deps")

SECRET_KEY = os.getenv("SECRET_KEY", "super_clave_secreta_de_respaldo")
ALGORITHM = "HS256"

REDIS_URL = os.getenv("REDIS_URL", "redis://redis:6379/0")
redis_limiter = redis.from_url(REDIS_URL, decode_responses=True)


class RateLimiter:
    """Escudo Anti-Fuerza Bruta optimizado. Limita por User ID si está logueado, sino por IP."""

    def __init__(self, max_requests: int, window_seconds: int):
        self.max_requests = max_requests
        self.window_seconds = window_seconds

    def __call__(self, request: Request):
        client_identifier = request.client.host if request.client else "127.0.0.1"

        token = request.cookies.get("access_token")
        if token:
            try:
                payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
                if payload.get("sub"):
                    client_identifier = f"user:{payload.get('sub')}"
            except InvalidTokenError:
                pass

        redis_key = f"rate_limit:{client_identifier}:{request.url.path}"

        try:
            current_requests = redis_limiter.incr(redis_key)
            if current_requests == 1:
                redis_limiter.expire(redis_key, self.window_seconds)

            if current_requests > self.max_requests:
                # Lanzamos 429 explícito
                raise HTTPException(
                    status_code=429,
                    detail=f"Límite de seguridad excedido. Máximo {self.max_requests} intentos cada {self.window_seconds} segundos.",
                )
        except redis.RedisError:
            # Programación Defensiva: Si Redis se cae, dejamos pasar el tráfico (Fail-open)
            # antes que tirar todo el servidor.
            pass


class GlobalRateLimiter:
    """Escudo Anti-DDoS Global. Corta la corriente si el servidor entero se satura."""

    def __init__(self, max_global_requests: int, window_seconds: int):
        self.max_requests = max_global_requests
        self.window_seconds = window_seconds

    def __call__(self, request: Request):
        global_key = f"global_limit:{request.url.path}"
        try:
            current_requests = redis_limiter.incr(global_key)
            if current_requests == 1:
                redis_limiter.expire(global_key, self.window_seconds)

            if current_requests > self.max_requests:
                raise HTTPException(
                    status_code=503,
                    detail="El servidor está experimentando un pico de tráfico anómalo. Por favor, espera unos segundos.",
                )
        except redis.RedisError:
            pass


def get_current_user(request: Request, session: Session = Depends(get_session)):
    """Verifica el JWT leyéndolo directamente desde la Cookie HttpOnly."""
    token = request.cookies.get("access_token")

    if not token:
        logger.debug(
            f"🛡️ [DEPS] Rechazado: Ruta {request.url.path} sin cookie 'access_token'."
        )
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED, detail="No autenticado"
        )

    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        username: str = payload.get("sub")

        redis_client.setex(f"user:online:{user.id}", 300, "1")

        if username is None:
            logger.warning("🛡️ [DEPS] Rechazado: Token válido pero sin campo 'sub'.")
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED, detail="No autenticado"
            )
    except jwt.ExpiredSignatureError:
        logger.debug(
            f"⏱️ [DEPS] Access Token expirado en ruta {request.url.path}. El frontend debería pedir /refresh."
        )
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED, detail="Token expirado"
        )
    except PyJWTError as e:

        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED, detail="No autenticado"
        )
    except Exception as e:
        pass

    user = session.exec(select(User).where(User.username == username)).first()
    if user is None:
        logger.error(f"🛡️ [DEPS] Rechazado: Usuario {username} en token pero no en DB.")
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED, detail="No autenticado"
        )

    return user

@router.post("/garbage-collect")
def run_garbage_collector(
    current_admin: User = Depends(get_current_admin),
    session: Session = Depends(get_session),
):
    """
    🧹 El Recolector de Basura:
    1. Borra Logs de más de 30 días.
    2. Borra Códigos de Invitación usados hace más de 30 días.
    3. Destruye archivos en /temp de R2 que tengan más de 24 horas.
    """
    logger.info(
        f"🧹 [GARBAGE_COLLECTOR] Protocolo iniciado por el Admin: {current_admin.username}"
    )

    now = datetime.now(timezone.utc)
    thirty_days_ago = now - timedelta(days=30)
    one_day_ago = now - timedelta(days=1)

    # --- 1. LIMPIEZA DE BASE DE DATOS (NEON) ---
    try:
        # Borrar Logs Viejos
        query_logs = delete(AuditLog).where(AuditLog.timestamp < thirty_days_ago)
        result_logs = session.exec(query_logs)
        deleted_logs = result_logs.rowcount

        # Borrar Invitaciones Usadas y Viejas
        query_invites = delete(InviteCode).where(
            InviteCode.is_used == True, InviteCode.created_at < thirty_days_ago
        )
        result_invites = session.exec(query_invites)
        deleted_invites = result_invites.rowcount

        session.commit()
    except Exception as e:
        session.rollback()
        logger.error(f"❌ [GARBAGE_COLLECTOR] Fallo crítico en la DB: {e}")
        raise HTTPException(status_code=500, detail="Fallo en la purga de PostgreSQL")

    # --- 2. LIMPIEZA DE INFRAESTRUCTURA (CLOUDFLARE R2) ---
    deleted_temp_files = 0
    try:
        # Listamos objetos en la carpeta temp/
        response = s3_client.list_objects_v2(
            Bucket=settings.R2_BUCKET_NAME, Prefix="temp/"
        )

        if "Contents" in response:
            objects_to_delete = []
            for obj in response["Contents"]:
                # Si el archivo lleva más de 24h sin ser movido a /prompts, es basura.
                if obj["LastModified"] < one_day_ago:
                    objects_to_delete.append({"Key": obj["Key"]})

            if objects_to_delete:
                # La API de S3/R2 permite borrar hasta 1000 objetos por petición HTTP
                for i in range(0, len(objects_to_delete), 1000):
                    batch = objects_to_delete[i : i + 1000]
                    s3_client.delete_objects(
                        Bucket=settings.R2_BUCKET_NAME, Delete={"Objects": batch}
                    )
                deleted_temp_files = len(objects_to_delete)

    except Exception as e:
        logger.error(f"❌ [GARBAGE_COLLECTOR] Fallo crítico en R2: {e}")
        # No lanzamos excepción aquí para que al menos la limpieza de DB sea exitosa
        pass

    logger.info(
        f"✅ [GARBAGE_COLLECTOR] Completado. Logs: {deleted_logs} | Invites: {deleted_invites} | R2_Temp: {deleted_temp_files}"
    )

    return {
        "status": "success",
        "deleted_logs": deleted_logs,
        "deleted_invites": deleted_invites,
        "deleted_temp_files": deleted_temp_files,
    }


@router.post("/users/action")
def manage_user_action(
    data: UserActionRequest,
    current_admin: User = Depends(get_current_admin),
    session: Session = Depends(get_session),
):
    target_user = session.get(User, data.user_id)
    if not target_user:
        raise HTTPException(status_code=404, detail="Usuario no encontrado")

    if target_user.role == "admin" and current_admin.id != target_user.id:
        raise HTTPException(
            status_code=403, detail="No puedes modificar a otro administrador."
        )

    if data.action == "upgrade":
        target_user.role = "artist"
        msg = f"Usuario ascendido a Artist VIP."

    elif data.action == "deactivate":
        target_user.is_active = False
        msg = f"Cuenta desactivada (Shadowban)."

    elif data.action == "activate":
        target_user.is_active = True
        msg = f"Cuenta reactivada."

    elif data.action == "reset_password":
        new_pass = secrets.token_urlsafe(6)
        target_user.hashed_password = get_password_hash(new_pass)
        session.commit()
        return {
            "status": "success",
            "message": "Contraseña reseteada",
            "new_password": new_pass,
        }

    else:
        raise HTTPException(status_code=400, detail="Acción no válida")

    session.add(target_user)
    session.commit()

    # 🟢 Registrar en Auditoría
    AuditService.log_action(
        session,
        current_admin.username,
        f"USER_{data.action.upper()}",
        f"Afectado: @{target_user.username}",
    )

    return {"status": "success", "message": msg}


@router.delete("/users/{user_id}")
def delete_user_completely(
    user_id: str,
    background_tasks: BackgroundTasks,
    current_admin: User = Depends(get_current_admin),
    session: Session = Depends(get_session),
):
    """
    Exterminio Total en Cascada.
    Respeta la integridad referencial de PostgreSQL (Foreign Keys).
    """
    target_user = session.get(User, user_id)
    if not target_user:
        raise HTTPException(status_code=404, detail="Usuario no encontrado")

    if target_user.role == "admin":
        raise HTTPException(
            status_code=403, detail="No se puede borrar a un administrador."
        )

    try:
        # 1. Recuperar todas las URLs de imágenes antes de borrar la DB
        prompts = session.exec(
            select(ImagePrompt).where(ImagePrompt.user_id == target_user.id)
        ).all()
        image_urls_to_delete = [p.image_url for p in prompts if p.image_url]

        # 2. Delegar borrado físico de R2 a Background Task
        if image_urls_to_delete:
            background_tasks.add_task(purge_r2_images_batch, image_urls_to_delete)

        # 3. 🟢 RESOLVER CONSTRAINT: Desvincular el Código de Invitación
        # No borramos el código para que siga constando como "usado", pero rompemos el enlace.
        invites = session.exec(
            select(InviteCode).where(InviteCode.used_by_id == target_user.id)
        ).all()
        for inv in invites:
            inv.used_by_id = None
            session.add(inv)

        # 4. 🟢 BORRADO EN CASCADA (Desde los hijos hasta el padre)
        # Borramos las métricas de galería que pertenezcan a él
        session.exec(
            delete(GalleryMetric).where(GalleryMetric.user_id == target_user.id)
        )

        # Borramos las imágenes
        session.exec(delete(ImagePrompt).where(ImagePrompt.user_id == target_user.id))

        # Borramos las colecciones
        session.exec(delete(Collection).where(Collection.user_id == target_user.id))

        # Finalmente, ejecutamos al usuario
        session.delete(target_user)

        # Consolidamos todos los cambios en una sola transacción ACID
        session.commit()

        AuditService.log_action(
            session,
            current_admin.username,
            "USER_DELETED",
            f"Exterminado: @{target_user.username} y {len(prompts)} prompts.",
        )

        return {
            "status": "success",
            "message": f"Usuario @{target_user.username} exterminado.",
        }

    except Exception as e:
        session.rollback()
        logger.error(f"Error crítico en exterminio de usuario: {e}")
        raise HTTPException(
            status_code=500, detail="Fallo al ejecutar el exterminio en cascada."
        )

    user_data: UserLogin, response: Response, session: Session = Depends(get_session)

    user_key = f"brute_force_user:{user_data.username}"
    attempts = redis_client.incr(user_key)

        redis_client.expire(user_key, 300)

        raise HTTPException(
            status_code=429,

        value=access_token, 
        max_age=60 * 15, 
        **COOKIE_PARAMS
    )
    
    response.set_cookie(
        key="refresh_token",
        value=refresh_token,
        max_age=60 * 60 * 24 * 7,
        **COOKIE_PARAMS,
    )

    AuditService.log_action(session, username, "LOGIN", "Inicio de sesión exitoso")

def create_unique_slug(name: str, user_id: int, session: Session) -> str:
    """Genera un slug seguro, quitando tildes y añadiendo número si hay colisión."""
    name_normalized = (
        unicodedata.normalize("NFKD", name).encode("ASCII", "ignore").decode("utf-8")
    )

    # 2. Limpiar caracteres raros y espacios por guiones
    base_slug = (
        re.sub(r"[^a-z0-9]+", "-", name_normalized.lower()).strip("-") or "coleccion"
    )

    slug = base_slug
    counter = 1

    # 3. Evitar colisiones (si ya tienes otra que se llame igual)
    while session.exec(
        select(Collection).where(Collection.user_id == user_id, Collection.slug == slug)
    ).first():
        slug = f"{base_slug}-{counter}"
        counter += 1

    return slug

@router.post("/collection/{collection_id}")
def descargar_coleccion_completa(
    collection_id: uuid.UUID,
    current_user: User = Depends(get_current_user),
    session: Session = Depends(get_session),
):
    """Delega la generación masiva del ZIP al Worker de Redis para no congelar la API."""

    col = session.get(Collection, collection_id)
    if not col:
        raise HTTPException(status_code=404, detail="Colección no encontrada")

    if col.is_private and col.user_id != current_user.id:
        raise HTTPException(status_code=403, detail="Esta colección es privada")

    cantidad_imagenes = session.exec(
        select(func.count(ImagePrompt.id)).where(
            ImagePrompt.collection_id == collection_id
        )
    ).one()

    limite_descarga = 50 if current_user.role == UserRole.VOYAGER else 1000

    if cantidad_imagenes > limite_descarga:
        raise HTTPException(
            status_code=403,
            detail=f"Tu plan '{current_user.role}' permite descargar un máximo de {limite_descarga} imágenes de golpe. Esta colección tiene {cantidad_imagenes}. ¡Haz upgrade a Artist!",
        )

    try:
        if current_user.role == UserRole.ARTIST:
            job = downloads_queue_premium.enqueue(
                generar_zip_coleccion,
                str(collection_id),
                current_user.email,
                job_timeout="1h",
            )
            queue_assigned = "premium"
        else:
            job = downloads_queue_free.enqueue(
                generar_zip_coleccion,
                str(collection_id),
                current_user.email,
                job_timeout="1h",
            )
            queue_assigned = "free"

        return {
            "status": "processing",
            "job_id": job.get_id(),
            "message": "Generando ZIP en segundo plano.",
            "queue": queue_assigned,
        }
    except Exception as e:
        raise HTTPException(
            status_code=500, detail="Error al encolar la descarga masiva"
        )


@router.get("/status/{job_id}")
def verificar_estado_descarga(
    job_id: str, current_user: User = Depends(get_current_user)
):
    """
    El frontend consulta este endpoint cada X segundos.
    Lee directamente de Redis el estado del Worker sin tocar la Base de Datos.
    """
    try:
        job = Job.fetch(job_id, connection=conn)

        if job.is_finished:
            return {"status": "finished", "download_url": job.result}
        elif job.is_failed:
            return {"status": "failed", "error": str(job.exc_info)}
        else:
            return {"status": "processing"}

    except NoSuchJobError:
        raise HTTPException(
            status_code=404, detail="El trabajo de descarga no existe o ha caducado"
        )
    except Exception as e:
        raise HTTPException(status_code=500, detail="Error interno al consultar Redis")

@router.post(
    "/generate-upload-url",
    dependencies=[Depends(RateLimiter(max_requests=100, window_seconds=60))],
)
def get_upload_url(
    data: PresignedUrlRequest, current_user: User = Depends(get_current_user)
):
    file_extension = data.file_name.split(".")[-1].lower()
    if file_extension not in ["png", "jpg", "jpeg", "webp"]:
        raise HTTPException(status_code=400, detail="Extensión no permitida.")

    safe_filename = f"{uuid.uuid4().hex}.{file_extension}"
    s3_key = f"temp/{current_user.id}/{safe_filename}"

    try:
        presigned_url = s3_client.generate_presigned_url(
            "put_object",
            Params={
                "Bucket": settings.R2_BUCKET_NAME,
                "Key": s3_key,
                "ContentType": f"image/{'jpeg' if file_extension == 'jpg' else file_extension}",
            },
            ExpiresIn=300,
        )

        base_url = settings.R2_PUBLIC_URL.rstrip("/")

        return {
            "status": "success",
            "upload_url": presigned_url,
            "key": s3_key,
            "file_url": f"{base_url}/{s3_key}",
        }
    except Exception as e:
        logger.error(f"Error AWS/R2: {e}")
        raise HTTPException(
            status_code=500, detail="Error generando acceso seguro al servidor."
        )


@router.post(
    "/save", dependencies=[Depends(RateLimiter(max_requests=100, window_seconds=60))]
)
def guardar_prompt(
    data: PromptCreate,
    current_user: User = Depends(get_current_user),
    session: Session = Depends(get_session),
):

    is_valid = validate_magic_bytes(data.image_url)

    base_url = settings.R2_PUBLIC_URL.rstrip("/") + "/"
    old_key = data.image_url.replace(base_url, "")

    if not old_key.startswith(f"temp/{current_user.id}/"):
        raise HTTPException(status_code=403, detail="No se puede guardar archivo")

    if not is_valid:
        delete_file(old_key)
        raise HTTPException(
            status_code=400, detail="El archivo está corrupto o es malicioso."
        )

    new_key = old_key.replace("temp/", "prompts/", 1)

    if not move_to_permanent(old_key, new_key):
        raise HTTPException(
            status_code=500, detail="Error interno al consolidar el archivo."
        )

    final_image_url = f"{base_url}{new_key}"
    data.image_url = final_image_url
    try:
        nuevo_prompt = ImagePrompt(
            user_id=current_user.id,
            title=data.title,
            prompt=data.prompt,
            negative_prompt=data.negative_prompt,
            software=data.software,
            model=data.model,
            sampler=data.sampler,
            steps=data.steps,
            cfg_scale=data.cfg_scale,
            seed=data.seed,
            resolution=data.resolution,
            image_url=data.image_url,
            extra_data=data.extra_data,
        )
        session.add(nuevo_prompt)
        session.commit()
        session.refresh(nuevo_prompt)

        return {
            "status": "success",
            "message": "Metadata guardada exitosamente",
            "data": nuevo_prompt,
        }
    except Exception as e:
        session.rollback()
        raise HTTPException(status_code=500, detail=str(e))

        query_prompt = query_prompt.where(User.id != current_user.id)

        query_col.order_by(Collection.created_at.desc()).limit(limit)
    ).all()
    prompts = session.exec(
        query_prompt.order_by(ImagePrompt.created_at.desc()).limit(limit)
    ).all()

    feed = []

    for col, user, items_count, cover_url in collections:
        feed.append(
            {
                "type": "collection",
                "id": f"col_{col.id}",
                "raw_id": col.id,
                "created_at": str(col.created_at),
                "title": col.name,
                "slug": col.slug,
                "username": user.username,
                "image_url": cover_url,
                "items_count": items_count or 0,
            }
        )

    for p, user in prompts:
        feed.append(
            {
                "type": "prompt",
                "id": f"prompt_{p.id}",
                "raw_id": p.id,
                "created_at": str(p.created_at),
                "title": p.title,
                "prompt": p.prompt,
                "username": user.username,
                "image_url": p.image_url,
                "model": p.model,
                "resolution": p.resolution,
            }
        )

    feed.sort(key=lambda x: x["created_at"], reverse=True)

    return {"status": "success", "feed": feed[:limit]}

import os
import jwt
import uuid
import bcrypt
from datetime import datetime, timedelta
from dotenv import load_dotenv

# Cargamos el .env
load_dotenv()

# Configuraciones del Token
SECRET_KEY = os.getenv("SECRET_KEY", "super_clave_secreta_de_respaldo")
ALGORITHM = "HS256"

# 🟢 Las nuevas duraciones de los tokens
ACCESS_TOKEN_EXPIRE_MINUTES = 15
REFRESH_TOKEN_EXPIRE_DAYS = 7


def get_password_hash(password: str) -> str:
    pwd_bytes = password.encode("utf-8")
    salt = bcrypt.gensalt()
    hashed_password = bcrypt.hashpw(pwd_bytes, salt)
    return hashed_password.decode("utf-8")


def verify_password(plain_password: str, hashed_password: str) -> bool:
    pwd_bytes = plain_password.encode("utf-8")
    hash_bytes = hashed_password.encode("utf-8")
    return bcrypt.checkpw(pwd_bytes, hash_bytes)


def create_access_token(data: dict) -> str:
    """Fabrica el Pase VIP (JWT) de 15 minutos"""
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
    to_encode.update({"exp": expire, "type": "access"})

    encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)
    return encoded_jwt


def create_refresh_token(data: dict, family_id: str = None) -> dict:
    """Fabrica el Pase Maestro Mutante de 7 días"""
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(days=REFRESH_TOKEN_EXPIRE_DAYS)

    # Generamos el ADN del token
    token_id = str(uuid.uuid4())
    token_family = family_id if family_id else str(uuid.uuid4())

    to_encode.update(
        {
            "exp": expire,
            "type": "refresh",
            "jti": token_id,
            "family": token_family,
        }
    )

    encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)

    # Devolvemos el JWT y sus IDs para que el router los guarde en Redis
    return {"token": encoded_jwt, "jti": token_id, "family_id": token_family}

def _get_chrome_options():

    options = Options()
    options.add_argument("--headless=new")
    options.add_argument("--no-sandbox")

    Context manager que levanta Chrome y garantiza su destrucción al terminar.
    Uso: with navegador_seguro() as driver: ...
    """

    try:
        yield driver
    finally:
        logger.info("🔒 Destruyendo instancia de Chrome para liberar RAM...")
        driver.quit()

from selenium.webdriver.common.by import By
from selenium.webdriver.support import expected_conditions as EC
from selenium.webdriver.support.ui import WebDriverWait

logger = logging.getLogger("analytics")

def extraer_metadata_galeria(driver, url: str, thumb_url: str) -> dict:
    """Extrae metadata y asocia la miniatura usando un driver existente."""
    # Abrimos una nueva pestaña para no perder la página de búsqueda
    driver.execute_script("window.open(arguments[0], '_blank');", url)
    driver.switch_to.window(driver.window_handles[-1])
    
    data = {"url": url, "thumb": thumb_url, "tags": {}, "title": "N/A", "pages": 0}
    wait = WebDriverWait(driver, 10)
    
    try:
        wait.until(EC.presence_of_element_located((By.ID, "gn")))
        data["title"] = driver.find_element(By.ID, "gn").text
        
        for row in driver.find_elements(By.CSS_SELECTOR, "#gdd tr"):
            if "Number of Pages" in row.text:
                data["pages"] = row.find_element(By.CLASS_NAME, "gdt2").text.split()[0]

        for row in driver.find_elements(By.CSS_SELECTOR, "#taglist tr"):
            cat = row.find_element(By.CLASS_NAME, "tc").text.replace(":", "").lower()
            tags = [a.text.lower() for a in row.find_elements(By.CSS_SELECTOR, "div a")]
            data["tags"][cat] = tags
    except Exception as e:
        logger.warning(f"⚠️ Falló la extracción en la galería (Posible aviso +18): {e}")
    finally:
        driver.close() # Cerramos la pestaña
        driver.switch_to.window(driver.window_handles[0]) # Volvems a la principal
    
    return data

def generar_resumen_analitico(db: list):
    """Genera un reporte analítico de los tags y artistas más populares."""
    all_artists = []
    all_tags = []
    for item in db:
        if "artist" in item.get("tags", {}):
            all_artists.extend(item["tags"]["artist"])
        for cat, tags in item.get("tags", {}).items():
            if cat != "artist": 
                all_tags.extend(tags)
                
    resumen = f"\n{'='*40}\n📊 RESUMEN ANALÍTICO\n{'='*40}\n"
    resumen += f"✅ Total Galerías analizadas: {len(db)}\n\n🎨 Top 5 Artistas:\n"
    for art, count in Counter(all_artists).most_common(5):
        resumen += f"   - {art.capitalize()}: {count} galerías\n"
        
    resumen += "\n🏷️ Top 5 Tags Relacionados:\n"
    for tag, count in Counter(all_tags).most_common(5):
        resumen += f"   - {tag}: {count} coincidencias\n"
    resumen += f"{'='*40}\n"
    
    logger.info(resumen)

from sqlmodel import Session, select
from datetime import datetime, timezone

from app.models.user import AuditLog
from app.models.gallery import GalleryMetric


class AuditService:
    @staticmethod
    def log_action(session: Session, username: str, action: str, details: str):
        """Registra una acción en el AuditLog de forma silenciosa."""
        try:
            log = AuditLog(username=username, action=action, details=details)
            session.add(log)
            session.commit()
        except Exception as e:
            # Nunca rompemos la app por un fallo en el log
            session.rollback()
            print(f"⚠️ Error al registrar auditoría: {e}")

    @staticmethod
    def log_gallery_interaction(
        session: Session, url: str, title: str, thumb: str, interaction_type: str
    ):
        """interaction_type puede ser 'view' o 'download'"""
        try:
            # Buscamos si la galería ya existe en nuestra base de datos
            metric = session.exec(
                select(GalleryMetric).where(GalleryMetric.url == url)
            ).first()

            if not metric:
                # Si no existe, la registramos por primera vez
                metric = GalleryMetric(url=url, title=title, thumb=thumb)
                session.add(metric)

            # Sumamos la métrica
            if interaction_type == "view":
                metric.views += 1
            elif interaction_type == "download":
                metric.downloads += 1

            metric.last_accessed = datetime.now(timezone.utc)

            session.add(metric)
            session.commit()

    @staticmethod
    def register_user(user_data, session: Session):
        """Lógica de negocio pura para registrar un usuario."""

        logger = logging.getLogger("auth_service")
        logger.info(f"📝 [AUTH_SERVICE] Intento de registro. Email: {user_data.email}")
        logger.info(f"🔑 [AUTH_SERVICE] Verificando código de invitación: {user_data.invite_code}")

        statement = (
            select(InviteCode)
            .where(InviteCode.code == user_data.invite_code)
            .where(InviteCode.is_used == False)
        )

        invite = session.exec(statement).first()

        if not invite:
            raise HTTPException(
                status_code=400, detail="Código de invitación inválido o ya usado"
            )

        # 2. Validación de usuario existente
        email_exists = session.exec(
            select(User).where(User.email == user_data.email)).first()
        
        if email_exists:
            raise HTTPException(status_code=400, detail="Este correo electrónico ya está registrado.")

        base_username = re.sub(r"[^a-z0-9]", "", user_data.email.split("@")[0].lower())
        if not base_username:
            base_username = (
                "user"  # Fallback extremo por si el email era ".-_@gmail.com"
            )

        username = base_username
        counter = 1

        while session.exec(select(User).where(User.username == username)).first():
            username = f"{base_username}{counter}"
            counter += 1

        # 4. Asignación de rol y creación
        assigned_role = "admin" if invite.code.startswith("TROYA-") else "voyager"

        new_user = User(
            email=user_data.email,
            username=username,
            hashed_password=get_password_hash(user_data.password),
            role=assigned_role,
        )

        session.add(new_user)
        session.commit()
        session.refresh(new_user)

        # 5. Quema de código
        invite.is_used = True
        invite.used_by_id = new_user.id
        session.add(invite)
        session.commit()

        logger.info(
            f"✨ ¡Usuario registrado! Email: {new_user.email} -> Username asignado: @{new_user.username}"
        )
        return new_user

    def rotate_tokens(refresh_token: str, session: Session, redis_client):
        logger = logging.getLogger("auth_service")
        logger.info("🔄 [AUTH_SERVICE] Iniciando secuencia de rotación de tokens...")

        if not refresh_token:

            )
            raise HTTPException(status_code=401, detail="No hay token de refresco")

        try:
            payload = jwt.decode(refresh_token, SECRET_KEY, algorithms=[ALGORITHM])
            username, jti, family_id = (
                payload.get("sub"),
                payload.get("jti"),
                payload.get("family"),
            )

            logger.info(
                f"🔍 [AUTH_SERVICE] Token decodificado -> User: {username} | JTI: {jti} | Family: {family_id}"
            )

            if not all([username, jti, family_id]) or payload.get("type") != "refresh":
                logger.error(
                    "❌ [AUTH_SERVICE] Rotación fallida: Estructura del token inválida o no es tipo 'refresh'."
                )
                raise HTTPException(
                    status_code=401, detail="Estructura de token inválida"
                )

            redis_key = f"session:{username}:{family_id}"
            active_jti = redis_client.get(redis_key)

            logger.info(
                f"📡 [AUTH_SERVICE] Consultando Redis -> Clave: {redis_key} | JTI Guardado: {active_jti}"
            )

            if active_jti and active_jti != jti:
                logger.warning(
                    f"🚨 [AUTH_SERVICE] ALERTA DE SEGURIDAD (Posible Robo o Doble Petición)."
                )
                logger.warning(f"   -> JTI Recibido: {jti}")
                logger.warning(f"   -> JTI en Redis: {active_jti}")
                redis_client.delete(redis_key)
                logger.info(
                    f"🧹 [AUTH_SERVICE] Sesión {family_id} destruida en Redis por seguridad."
                )
                raise HTTPException(
                    status_code=403, detail="ALERTA: Robo de sesión detectado."
                )

            if not active_jti:
                logger.warning(
                    f"⚠️ [AUTH_SERVICE] Rotación rechazada: La sesión {family_id} ya expiró o fue cerrada remotamente."
                )
                raise HTTPException(
                    status_code=401, detail="Sesión expirada o cerrada remotamente."
                )

            user = session.exec(select(User).where(User.username == username)).first()
            if not user:
                logger.error(
                    f"❌ [AUTH_SERVICE] El usuario {username} existe en el token pero no en la Base de Datos."
                )
                raise HTTPException(status_code=401, detail="Usuario no encontrado")

            # Generación de nuevos tokens
            new_access_token = create_access_token(
                data={"sub": username, "role": user.role}
            )
            new_refresh_data = create_refresh_token(
                data={"sub": username}, family_id=family_id
            )

            # Actualizamos Redis
            redis_client.setex(redis_key, 60 * 60 * 24 * 7, new_refresh_data["jti"])

            logger.info(
                f"✅ [AUTH_SERVICE] Rotación EXITOSA. Nuevo JTI asignado: {new_refresh_data['jti']}"
            )

            return new_access_token, new_refresh_data["token"]

        except jwt.ExpiredSignatureError:
            logger.warning("⏱️ [AUTH_SERVICE] El Refresh Token ha caducado por tiempo.")
            raise HTTPException(
                status_code=401, detail="Token expirado. Inicia sesión."
            )
        except jwt.JWTError as e:

            raise HTTPException(status_code=401, detail="Firma inválida.")

def generar_zip_coleccion(collection_id: str, user_email: str):
    """
    Tarea asíncrona para RQ. Empaqueta una colección entera en un ZIP
    usando buffers de memoria y la sube de vuelta a R2 como un enlace temporal.
    """
    s3_client = boto3.client(
        "s3",
        endpoint_url=settings.R2_ENDPOINT_URL,
        aws_access_key_id=settings.R2_ACCESS_KEY_ID,
        aws_secret_access_key=settings.R2_SECRET_ACCESS_KEY,
        region_name="auto",
    )

    with Session(engine) as session:
        coleccion = session.get(Collection, collection_id)
        if not coleccion:
            return

        prompts = session.exec(
            select(ImagePrompt).where(ImagePrompt.collection_id == collection_id)
        ).all()

    # Buffer en memoria para no usar el disco SSD del servidor
    zip_buffer = io.BytesIO()
    metadata_list = []

    with zipfile.ZipFile(zip_buffer, "w", zipfile.ZIP_DEFLATED) as zip_file:
        for idx, prompt in enumerate(prompts):
            # 1. Extraer key de R2 desde la URL pública
            file_key = prompt.image_url.split("/")[-1]
            if "prompts/" not in file_key:
                file_key = f"prompts/{file_key}"

            # 2. Descargar imagen desde R2 a memoria
            try:
                response = s3_client.get_object(
                    Bucket=settings.R2_BUCKET_NAME, Key=file_key
                )
                image_bytes = response["Body"].read()

                # 3. Escribir imagen en el ZIP
                filename = f"{idx:03d}_{prompt.title.replace(' ', '_')}.png"
                zip_file.writestr(filename, image_bytes)

                # 4. Guardar metadatos para el JSON final
                metadata_list.append(
                    {
                        "filename": filename,
                        "prompt": prompt.prompt,
                        "model": prompt.model,
                        "sampler": prompt.sampler,
                    }
                )
            except Exception:
                continue  # Tolerancia a fallos: si una imagen falla, el ZIP sigue

        # 5. Escribir el manifiesto JSON con todos los prompts
        zip_file.writestr("prompts_metadata.json", json.dumps(metadata_list, indent=2))

    zip_buffer.seek(0)

    # 6. Subir el ZIP temporal a R2
    zip_key = f"temp/downloads/{collection_id}.zip"
    s3_client.upload_fileobj(
        zip_buffer,
        settings.R2_BUCKET_NAME,
        zip_key,
        ExtraArgs={"ContentType": "application/zip"},
    )

    # 7. Generar URL pre-firmada (caduca en 1 hora)
    download_url = s3_client.generate_presigned_url(
        "get_object",
        Params={"Bucket": settings.R2_BUCKET_NAME, "Key": zip_key},
        ExpiresIn=3600,
    )

    # (Opcional) Aquí notificarías al usuario vía WebSockets o Email de que su ZIP está listo
    return download_url

MAGIC_SIGNATURES = {
    "png": b"\x89PNG\r\n\x1a\n",
    "jpeg": b"\xff\xd8\xff",
    "webp_riff": b"RIFF",
    "webp_webp": b"WEBP",
}


def validate_magic_bytes(file_url: str) -> bool:
    """
    Verifica la autenticidad del archivo descargando solo 256 bytes.
    Programación Defensiva: Asume falsedad por defecto.
    """
    if not file_url:
        return False

    if not file_url.startswith(settings.R2_PUBLIC_URL):
        logger.error(
            f"[SECURITY] Intento de SSRF bloqueado: {file_url}"
        )
        return False

    try:
        # Inyectamos cabecera Range de forma dinámica para esta petición
        response = http_session.get(
            file_url, headers={"Range": "bytes=0-256"}, timeout=5
        )
        response.raise_for_status()
        header_bytes = response.content

        if header_bytes.startswith(MAGIC_SIGNATURES["png"]):
            return True
        if header_bytes.startswith(MAGIC_SIGNATURES["jpeg"]):
            return True
        if (
            header_bytes.startswith(MAGIC_SIGNATURES["webp_riff"])
            and header_bytes[8:12] == MAGIC_SIGNATURES["webp_webp"]
        ):
            return True

        logger.warning(
            f"[SECURITY] Archivo malicioso detectado y bloqueado. URL: {file_url}"
        )
        return False

    except requests.RequestException as e:
        logger.error(f"[SECURITY] Error de red al auditar URL {file_url}: {e}")
        return False

    """Mueve un objeto dentro del bucket y borra el original."""

        s3_client.copy_object(
            Bucket=settings.R2_BUCKET_NAME,

            Key=new_key,
        )

    except Exception as e:

# backend/app/utils/storage.py
import json
import logging
import os
import boto3
from botocore.exceptions import ClientError

logger = logging.getLogger("storage")

R2_ENDPOINT_URL = os.getenv("R2_ENDPOINT_URL")
R2_ACCESS_KEY = os.getenv("R2_ACCESS_KEY")
R2_SECRET_KEY = os.getenv("R2_SECRET_KEY")
R2_BUCKET_NAME = os.getenv("R2_BUCKET_NAME")
R2_PUBLIC_URL = os.getenv("R2_PUBLIC_URL")

# Inicializamos el cliente S3 de forma segura (solo si hay credenciales)
s3_client = None
if all([R2_ENDPOINT_URL, R2_ACCESS_KEY, R2_SECRET_KEY]):
    s3_client = boto3.client(
        "s3",
        endpoint_url=R2_ENDPOINT_URL,
        aws_access_key_id=R2_ACCESS_KEY,
        aws_secret_access_key=R2_SECRET_KEY,
        region_name="auto",  # Cloudflare R2 requiere "auto" o "us-east-1"
    )


def subir_imagen_a_r2(
    file_bytes: bytes, file_name: str, content_type: str = "image/webp"
) -> str:
    """
    Sube un archivo binario a Cloudflare R2 y devuelve la URL pública final.
    """
    if not s3_client or not R2_BUCKET_NAME:
        logger.error("⚠️ Credenciales de R2 no configuradas. Abortando subida.")
        raise ValueError("Servicio de almacenamiento en la nube no configurado.")

    try:
        logger.info(f"☁️ Subiendo {file_name} a R2...")
        s3_client.put_object(
            Bucket=R2_BUCKET_NAME,
            Key=file_name,
            Body=file_bytes,
            ContentType=content_type,
        )

        # Construir y devolver la URL pública exacta para guardarla en PostgreSQL
        url_publica = f"{R2_PUBLIC_URL.rstrip('/')}/{file_name}"
        logger.info(f"✅ Subida exitosa: {url_publica}")
        return url_publica

    except ClientError as e:
        logger.error(f"❌ Error de conexión con R2 (Boto3): {e}")
        raise e
    except Exception as e:
        logger.error(f"❌ Error inesperado subiendo a R2: {e}")
        raise e


def generar_presigned_url(file_name: str, content_type: str) -> dict:
    """Genera un ticket de un solo uso para que el Frontend suba directo a R2"""
    if not s3_client or not R2_BUCKET_NAME:
        raise ValueError("Servicio R2 no configurado en .env.")
    try:
        # Genera una URL temporal con permiso explícito de 'put_object' (Subida)
        presigned_url = s3_client.generate_presigned_url(
            "put_object",
            Params={
                "Bucket": R2_BUCKET_NAME,
                "Key": file_name,
                "ContentType": content_type,
            },
            ExpiresIn=3600,  # El ticket caduca en 1 hora
        )

        # Esta será la URL final pública una vez subido el archivo
        url_publica = f"{R2_PUBLIC_URL.rstrip('/')}/{file_name}"
        return {"upload_url": presigned_url, "file_url": url_publica}

    except Exception as e:
        logger.error(f"❌ Error generando presigned URL: {e}")
        raise e


def eliminar_de_r2(file_url: str):
    """Borra físicamente el archivo del bucket para no dejar huérfanos"""
    if not s3_client or not R2_BUCKET_NAME or not file_url:
        return
    try:
        base_url = R2_PUBLIC_URL.rstrip("/")
        # Verificamos que la URL pertenezca a nuestro bucket
        if file_url.startswith(base_url):
            # Extraemos la ruta interna (ej. prompts/edu/archivo.png)
            key = file_url.replace(base_url + "/", "")
            logger.info(f"🗑️ Eliminando de R2: {key}")
            s3_client.delete_object(Bucket=R2_BUCKET_NAME, Key=key)
    except Exception as e:
        logger.error(f"❌ Error eliminando objeto de R2: {e}")

def preparar_directorio(ruta_base: str, nombre_carpeta: str) -> str:
    """Crea el directorio de forma segura si no existe."""
    ruta_completa = os.path.join(ruta_base, nombre_carpeta)
    os.makedirs(ruta_completa, exist_ok=True)
    return ruta_completa


def guardar_archivo(ruta_directorio: str, nombre_archivo: str, iter_content) -> str:
    """Guarda un flujo de bytes (stream) en el disco duro local."""
    ruta_archivo = os.path.join(ruta_directorio, nombre_archivo)
    try:
        with open(ruta_archivo, "wb") as f:
            for chunk in iter_content:
                f.write(chunk)
        return ruta_archivo
    except Exception as e:
        logger.error(f"❌ Error al escribir en disco ({ruta_archivo}): {e}")
        raise e


def guardar_json(ruta_directorio: str, nombre_archivo: str, datos: list | dict) -> str:
    """Guarda un diccionario o lista de Python como archivo .json local"""
    os.makedirs(ruta_directorio, exist_ok=True)
    ruta_archivo = os.path.join(ruta_directorio, nombre_archivo)
    try:
        with open(ruta_archivo, "w", encoding="utf-8") as f:
            json.dump(datos, f, indent=4, ensure_ascii=False)
        return ruta_archivo
    except Exception as e:
        logger.error(f"❌ Error al guardar JSON ({ruta_archivo}): {e}")
        raise e

def ejecutar_mapeo(termino: str, paginas: int):
    # CORREGIDO: URL completa restaurada
    url_base = f"https://e-hentai.org/?f_search={quote(termino)}"
    base_de_datos = []

    with navegador_seguro() as driver:
        wait = WebDriverWait(driver, 10)

        for p in range(paginas):
            logger.info(f"\n🌐 Cargando URL: {url_base}&page={p}")
            driver.get(f"{url_base}&page={p}")

            try:
                wait.until(EC.presence_of_element_located((By.CLASS_NAME, "itg")))
                enlaces_galerias = driver.find_elements(
                    By.CSS_SELECTOR, ".itg a[href*='/g/']"
                )

                links_unicos = []
                for enlace in enlaces_galerias:
                    href = enlace.get_attribute("href")
                    if href and href not in links_unicos:
                        links_unicos.append(href)

                if not links_unicos:
                    logger.info("ℹ️ No se encontraron más galerías.")
                    break

                logger.info(
                    f"📦 Se encontraron {len(links_unicos)} galerías. Iniciando escaneo..."
                )

                for link in links_unicos:
                    try:
                        elemento = driver.find_element(
                            By.CSS_SELECTOR, f"a[href='{link}']"
                        )
                        thumb_url = "N/A"
                        try:
                            # CORREGIDO: XPath restaurado para encontrar la imagen
                            thumb_url = elemento.find_element(
                                By.XPATH, ".//img | ../..//img"
                            ).get_attribute("src")
                        except:
                            pass

                        logger.info(f"🔎 Analizando: {link.split('/')[-3]}")
                        info = extraer_metadata_galeria(driver, link, thumb_url)
                        base_de_datos.append(info)
                        time.sleep(2)  # Aumentado para mayor seguridad
                    except Exception as e:
                        logger.error(f"⚠️ Error en enlace {link}: {e}")
            except Exception:
                logger.error("🚨 ERROR CRÍTICO: No se encontró la tabla. ¿Captcha?")
                break

    if base_de_datos:
        generar_resumen_analitico(base_de_datos)
        nombre_archivo = f"analytics_{termino.replace(' ', '_').lower()}.json"
        ruta = guardar_json(
            "/storage/galleries" if os.path.exists("/storage") else ".",
            nombre_archivo,
            base_de_datos,
        )
        logger.info(f"📁 Datos guardados en: {ruta}")


if __name__ == "__main__":
    print("\n" + "=" * 40 + "\n🤖 HERRAMIENTA DE ANALÍTICAS\n" + "=" * 40)

from unittest.mock import patch
from fastapi.testclient import TestClient
from sqlmodel import Session
from app.models.user import User
from app.models.collection import Collection
from app.models.prompt import ImagePrompt
from app.core.config import settings


# =====================================================================
# 🛡️ 1. ATAQUES DE FUERZA BRUTA (RATE LIMITING)
# =====================================================================
def test_brute_force_login_rate_limiter(client: TestClient):
    """Test Defensivo: El escudo de Redis corta la cabeza al sexto intento."""
    for _ in range(5):
        res = client.post(
            "/api/auth/login", json={"username": "hacker_bot", "password": "123"}
        )
        assert res.status_code == 401

    res_bloqueado = client.post(
        "/api/auth/login", json={"username": "hacker_bot", "password": "123"}
    )

    assert res_bloqueado.status_code == 429
    # 🟢 FIX 1: Esperamos el mensaje de la dependencia de RateLimiter global
    assert "límite de seguridad" in res_bloqueado.json().get("detail", "").lower()


# =====================================================================
# 🛡️ 2. EVASIÓN DE FIRMAS MÁGICAS (ARCHIVOS MALICIOSOS)
# =====================================================================
def test_malicious_file_upload_magic_bytes(client: TestClient, normal_user: User):
    """Test Defensivo: Bloqueo de ejecutables disfrazados de imágenes"""
    login_res = client.post(
        "/api/auth/login",
        json={"username": "test_voyager", "password": "PassSegura123!"},
    )
    client.cookies = login_res.cookies

    # 🟢 FIX 2: Usamos el R2_PUBLIC_URL dinámico para pasar el primer filtro de seguridad
    base_url = settings.R2_PUBLIC_URL.rstrip("/")
    payload = {
        "title": "Gatito bonito",
        "prompt": "Un gatito cyberpunk",
        "image_url": f"{base_url}/temp/{normal_user.id}/virus.png",
    }

    with patch("app.api.routes_prompt.validate_magic_bytes", return_value=False), patch(
        "app.api.routes_prompt.delete_file"
    ) as mock_delete:

        response = client.post("/api/prompts/save", json=payload)

        assert response.status_code == 400
        assert "corrupto o es malicioso" in response.json().get("detail", "")
        mock_delete.assert_called_once()


# =====================================================================
# 🛡️ 3. AUTORIZACIÓN ROTA (BOLA / IDOR)
# =====================================================================
def test_bola_private_collection_theft(
    client: TestClient, session: Session, admin_user: User, normal_user: User
):
    """Test Defensivo: 'Voyager' intenta descargar la colección oculta del 'Admin'"""
    private_col = Collection(
        user_id=admin_user.id,
        name="Secretos de Estado",
        slug="secretos",
        is_private=True,
    )
    session.add(private_col)
    session.commit()

    login_res = client.post(
        "/api/auth/login",
        json={"username": "test_voyager", "password": "PassSegura123!"},
    )
    client.cookies = login_res.cookies

    response = client.post(f"/api/download/collection/{private_col.id}")

    assert response.status_code == 403
    assert "privada" in response.json().get("detail", "").lower()


# =====================================================================
# 🛡️ 4. EXPLOTACIÓN LÍMITES DE NEGOCIO (QUOTA BYPASS)
# =====================================================================
def test_voyager_plan_limit_exploitation(
    client: TestClient, session: Session, normal_user: User
):
    """
    Test Defensivo: El plan Voyager permite descargar máx 50 imágenes.
    Verificamos que el sistema protege el modelo de negocio.
    """
    public_col = Collection(
        user_id=normal_user.id, name="Masiva", slug="masiva", is_private=False
    )
    session.add(public_col)
    session.commit()

    # 🟢 FIX 3: Pragmatismo Radical. En lugar de Mockear y romper la sesión,
    # insertamos 51 imágenes en la memoria RAM (Tarda 0.005 segundos).
    fake_prompts = [
        ImagePrompt(
            user_id=normal_user.id,
            collection_id=public_col.id,
            title=f"Imagen {i}",
            prompt="test",
        )
        for i in range(51)
    ]
    session.add_all(fake_prompts)
    session.commit()

    login_res = client.post(
        "/api/auth/login",
        json={"username": "test_voyager", "password": "PassSegura123!"},
    )
    client.cookies = login_res.cookies

    # Intentamos descargar la colección
    response = client.post(f"/api/download/collection/{public_col.id}")

    assert response.status_code == 403
    assert "máximo de 50 imágenes" in response.json().get("detail", "").lower()

# Neon Serverless Postgres

Neon is a serverless Postgres platform that separates compute and storage to offer autoscaling, branching, instant restore, and scale-to-zero. It's fully compatible with Postgres and works with any language, framework, or ORM that supports Postgres.

## Neon Documentation

The Neon documentation is the source of truth for all Neon-related information. Always verify claims against the official docs before responding. Neon features and APIs evolve, so prefer fetching current docs over relying on training data.

### Fetching Docs as Markdown

Any Neon doc page can be fetched as markdown in two ways:

1. **Append `.md` to the URL** (simplest): https://neon.com/docs/introduction/branching.md
2. **Request `text/markdown`** on the standard URL: `curl -H "Accept: text/markdown" https://neon.com/docs/introduction/branching`

Both return the same markdown content. Use whichever method your tools support.

The docs index lists every available page with its URL and a short description:

## Branching

Use this when the user is planning isolated environments, schema migration testing, preview deployments, or branch lifecycle automation.

Key points:

- Branches are instant, copy-on-write clones (no full data copy).
- Each branch has its own compute endpoint.
- Use the neonctl CLI or MCP server to create, inspect, and compare branches.

Link: https://neon.com/docs/ai/skills/neon-postgres/references/branching.md

## Autoscaling

## Scale to Zero

Use this when optimizing idle costs and discussing suspend/resume behavior, including cold-start trade-offs.

Key points:

- Idle computes suspend automatically (default 5 minutes, configurable) (unless disabled - launch & scale plan only)
- First query after suspend typically has a cold-start penalty (around hundreds of ms)
- Storage remains active while compute is suspended.

Link: https://neon.com/docs/introduction/scale-to-zero.md

## Instant Restore

Use this when the user needs point-in-time recovery or wants to restore data state without traditional backup restore workflows.

Key points:

- Restore windows depend on plan limits.
- Users can create branches from historical points-in-time.
- Time Travel queries can be used for historical inspection workflows.

Link: https://neon.com/docs/introduction/branch-restore.md

## Read Replicas

Use this for read-heavy workloads where the user needs dedicated read-only compute without duplicating storage.

Key points:

- Replicas are read-only compute endpoints sharing the same storage.
- Creation is fast and scaling is independent from primary compute.
- Typical use cases: analytics, reporting, and read-heavy APIs.

Link: https://neon.com/docs/introduction/read-replicas.md

## Connection Pooling

Use this when the user is in serverless or high-concurrency environments and needs safe, scalable Postgres connection management.

Key points:

- Neon pooling uses PgBouncer.
- Add `-pooler` to endpoint hostnames to use pooled connections.
- Pooling is especially important in serverless runtimes with bursty concurrency.

Link: https://neon.com/docs/connect/connection-pooling.md

## Logical Replication

Use this when integrating CDC pipelines, external Postgres sync, or replication-based data movement.

Key points:
