from sqlmodel import SQLModel, Session, text

    print("🚨 [1/4] Borrando y recreando el esquema 'public'...")
    with Session(engine) as session:
        session.exec(text("DROP SCHEMA public CASCADE;"))
        session.exec(text("CREATE SCHEMA public;"))
        session.commit()

    print("🧩 [2/4] Activando extensión pgvector...")
    with Session(engine) as session:
        session.exec(text("CREATE EXTENSION IF NOT EXISTS vector;"))
        session.commit()

    SQLModel.metadata.create_all(engine)

    print("✅ [4/4] ¡Base de Datos Neon alineada y lista!")

    reset_database()

import logging
import time

from rq import Queue, Worker

from sqlmodel import Session, select

from app.models.user import User
from app.models.directory import Directory
from app.models.prompt import ImagePrompt

from app.core.db import engine
from app.core.redis_config import conn

from app.services.embedding_service import EmbeddingService
from app.services.wd14_service import WD14Tagger

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(message)s")
logger = logging.getLogger("rq_worker")


def process_image_ai(prompt_id: str):
    with Session(engine) as session:
        prompt = session.get(ImagePrompt, prompt_id)
        if not prompt:
            logger.error(f"❌ [WORKER] Prompt {prompt_id} no encontrado en DB.")
            return

        try:
            logger.info(f"🧠 [WORKER] Iniciando procesamiento de IA para: {prompt_id}")
            texto_a_vectorizar = prompt.prompt or prompt.title or "Generative Art"
            start_time_emb = time.time()

            vector = EmbeddingService.generate_vector(prompt.prompt or "Generative Art")
            prompt.embedding = vector

            tiempo_emb = round(time.time() - start_time_emb, 2)

            logger.info(f"🏷️ [WORKER] Extrayendo etiquetas visuales...")
            start_time_tags = time.time()

            wd14_manager = WD14Tagger()

            etiquetas = wd14_manager.get_tags(prompt.image_url)
            prompt.auto_tags = etiquetas

            tiempo_tags = round(time.time() - start_time_tags, 2)

            session.add(prompt)
            session.commit()
            
            logger.info(f"✅ [WORKER] IA completada con éxito para ID: {prompt_id}")

        except Exception as e:
            logger.error(
                f"❌ [WORKER] Fallo crítico procesando IA en ID {prompt_id}: {e}"
            )


if __name__ == "__main__":
    try:
        queues = [
            Queue("default", connection=conn),
            Queue("downloads_premium", connection=conn),
            Queue("downloads_free", connection=conn),
        ]

        worker = Worker(queues, connection=conn)
        logger.info("👷 Worker RQ Online. Escuchando colas: AI + Descargas.")
        worker.work()
    except Exception as e:
        logger.error(f"❌ El Worker ha muerto: {e}")

clean_url = (
    raw_url.strip().strip("'").strip('"').replace("postgres://", "postgresql://", 1)

# Asegurar sslmode para Neon

    clean_url += f"{separator}sslmode=require"

# Interpret the config file for Python logging.

    fileConfig(config.config_file_name)

# 4. Indicar a Alembic dónde está el esquema de metadatos
target_metadata = SQLModel.metadata


def run_migrations_offline() -> None:
    """Run migrations in 'offline' mode.

    This configures the context with just a URL
    and not an Engine, though an Engine is acceptable
    here as well.  By skipping the Engine creation
    we don't even need a DBAPI to be available.

    Calls to context.execute() here emit the given string to the
    script output.

    """
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )

    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online() -> None:
    """Run migrations in 'online' mode.

    In this scenario we need to create an Engine

def run_network_diagnostics():
    logger.info("============== 🕵️ INICIANDO DIAGNÓSTICO DE RED ==============")
    parsed = urlparse(clean_url)
    host = parsed.hostname
    port = parsed.port or 5432

    logger.info(f"📡 [PROBE 1/3] Resolviendo DNS para: {host}")
    try:
        ip = socket.gethostbyname(host)
        logger.info(f"✅ [PROBE 1/3] IP Resuelta: {ip}")
    except Exception as e:
        logger.error(f"❌ [PROBE 1/3] Fallo de resolución DNS: {e}")
        return False

    logger.info(f"🚪 [PROBE 2/3] Probando puerto TCP {ip}:{port}...")
    try:
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.settimeout(5.0)
        sock.connect((ip, port))
        logger.info("✅ [PROBE 2/3] Puerto TCP abierto y accesible.")
        sock.close()
    except Exception as e:
        logger.error(f"❌ [PROBE 2/3] El puerto está bloqueado o inaccesible: {e}")
        return False

    logger.info("🔐 [PROBE 3/3] Probando Handshake SSL y Autenticación directa...")
    try:
        start_time = time.time()
        conn = psycopg2.connect(clean_url, connect_timeout=10)
        elapsed = round(time.time() - start_time, 2)
        logger.info(f"✅ [PROBE 3/3] ¡Autenticación SSL exitosa en {elapsed}s!")

        cur = conn.cursor()
        cur.execute("SELECT version();")
        version = cur.fetchone()
        logger.info(f"🐘 [PROBE] Versión de DB: {version[0][:30]}...")
        cur.close()
        conn.close()
        logger.info("============== 🕵️ DIAGNÓSTICO COMPLETADO ==============\n")
        return True
    except Exception as e:
        elapsed = round(time.time() - start_time, 2)
        logger.error(f"❌ [PROBE 3/3] Falló la negociación SSL/Auth tras {elapsed}s.")
        logger.error(f"Detalle exacto: {e}")
        logger.info("============== 🕵️ DIAGNÓSTICO COMPLETADO ==============\n")
        return False


def init_db():
    logger.info("🔌 [DB] Verificando infraestructura antes de iniciar...")

    run_network_diagnostics()

    max_retries = 3
    retry_delay = 5

    logger.info(
        "🔌 [DB] Iniciando secuencia de sincronización de tablas (SQLAlchemy)..."
    )

    for attempt in range(1, max_retries + 1):
        try:
            logger.info(
                f"⏳ [DB] [Intento {attempt}/{max_retries}] Sincronizando metadata..."
            )

            SQLModel.metadata.create_all(engine)
            logger.info("✅ [DB] ¡Metadata sincronizada correctamente!")

            with Session(engine) as session:
                existing_code = session.exec(select(InviteCode)).first()
                if not existing_code:
                    master_code = f"TROYA-{secrets.token_hex(4).upper()}"
                    new_invite = InviteCode(code=master_code)

class CorrelationIdMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        correlation_id = request.headers.get("X-Request-ID", str(uuid.uuid4()))

        request.state.correlation_id = correlation_id

        start_time = time.time()
        response = await call_next(request)
        process_time = time.time() - start_time

        response.headers["X-Request-ID"] = correlation_id
        response.headers["X-Process-Time"] = str(process_time)
        return response

import os
import redis
import logging
import jwt
from fastapi import Depends, HTTPException, status, Request
from jwt.exceptions import InvalidTokenError, PyJWTError
from sqlmodel import Session, select
from app.core.redis_config import conn as redis_client
from app.core.db import get_session
from app.models.user import User

logger = logging.getLogger("deps")

SECRET_KEY = os.getenv("SECRET_KEY", "super_clave_secreta_de_respaldo")
ALGORITHM = "HS256"

REDIS_URL = os.getenv("REDIS_URL", "redis://redis:6379/0")
redis_limiter = redis.from_url(REDIS_URL, decode_responses=True)


class RateLimiter:
    """Escudo Anti-Fuerza Bruta optimizado. Limita por User ID si está logueado, sino por IP."""

    def __init__(self, max_requests: int, window_seconds: int):
        self.max_requests = max_requests
        self.window_seconds = window_seconds

    def __call__(self, request: Request):
        client_identifier = request.client.host if request.client else "127.0.0.1"

        token = request.cookies.get("access_token")
        if token:
            try:
                payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
                if payload.get("sub"):
                    client_identifier = f"user:{payload.get('sub')}"
            except InvalidTokenError:
                pass

        redis_key = f"rate_limit:{client_identifier}:{request.url.path}"

        try:
            current_requests = redis_limiter.incr(redis_key)
            if current_requests == 1:
                redis_limiter.expire(redis_key, self.window_seconds)

            if current_requests > self.max_requests:
                # Lanzamos 429 explícito
                raise HTTPException(
                    status_code=429,
                    detail=f"Límite de seguridad excedido. Máximo {self.max_requests} intentos cada {self.window_seconds} segundos.",
                )
        except redis.RedisError:
            # Programación Defensiva: Si Redis se cae, dejamos pasar el tráfico (Fail-open)
            # antes que tirar todo el servidor.
            pass


class GlobalRateLimiter:
    """Escudo Anti-DDoS Global. Corta la corriente si el servidor entero se satura."""

    def __init__(self, max_global_requests: int, window_seconds: int):
        self.max_requests = max_global_requests
        self.window_seconds = window_seconds

    def __call__(self, request: Request):
        global_key = f"global_limit:{request.url.path}"
        try:
            current_requests = redis_limiter.incr(global_key)
            if current_requests == 1:
                redis_limiter.expire(global_key, self.window_seconds)

            if current_requests > self.max_requests:
                raise HTTPException(
                    status_code=503,
                    detail="El servidor está experimentando un pico de tráfico anómalo. Por favor, espera unos segundos.",
                )
        except redis.RedisError:
            pass


def get_current_user(request: Request, session: Session = Depends(get_session)):
    """Verifica el JWT leyéndolo directamente desde la Cookie HttpOnly."""
    token = request.cookies.get("access_token")

    if not token:
        logger.debug(
            f"🛡️ [DEPS] Rechazado: Ruta {request.url.path} sin cookie 'access_token'."
        )
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED, detail="No autenticado"
        )

    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        username: str = payload.get("sub")

        redis_client.setex(f"user:online:{user.id}", 300, "1")

        if username is None:
            logger.warning("🛡️ [DEPS] Rechazado: Token válido pero sin campo 'sub'.")
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED, detail="No autenticado"
            )
    except jwt.ExpiredSignatureError:
        logger.debug(
            f"⏱️ [DEPS] Access Token expirado en ruta {request.url.path}. El frontend debería pedir /refresh."
        )
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED, detail="Token expirado"
        )
    except PyJWTError as e:
        logger.warning(f"🛡️ [DEPS] Rechazado: Error decodificando JWT: {str(e)}")
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED, detail="No autenticado"
        )
    except Exception as e:
        pass

    user = session.exec(select(User).where(User.username == username)).first()
    if user is None:
        logger.error(f"🛡️ [DEPS] Rechazado: Usuario {username} en token pero no en DB.")
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED, detail="No autenticado"
        )

    return user


def get_current_admin(current_user: User = Depends(get_current_user)):
    """Dependencia de RBAC: Verifica que el usuario logueado sea Administrador."""
    if current_user.role != "admin":
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="No tienes suficientes privilegios (Requiere rol Admin)",
        )
    return current_user


def get_optional_user(request: Request, session: Session = Depends(get_session)):
    """Verifica el JWT pero no lanza error si no existe. Útil para identificar al dueño en rutas públicas."""
    token = request.cookies.get("access_token")
    if not token:
        return None
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        username: str = payload.get("sub")
        if username:
            return session.exec(select(User).where(User.username == username)).first()
    except PyJWTError:
        return None
    return None

@router.post("/garbage-collect")
def run_garbage_collector(
    current_admin: User = Depends(get_current_admin),
    session: Session = Depends(get_session),
):
    """
    🧹 El Recolector de Basura:
    1. Borra Logs de más de 30 días.
    2. Borra Códigos de Invitación usados hace más de 30 días.
    3. Destruye archivos en /temp de R2 que tengan más de 24 horas.
    """
    logger.info(
        f"🧹 [GARBAGE_COLLECTOR] Protocolo iniciado por el Admin: {current_admin.username}"
    )

    now = datetime.now(timezone.utc)
    thirty_days_ago = now - timedelta(days=30)
    one_day_ago = now - timedelta(days=1)

    # --- 1. LIMPIEZA DE BASE DE DATOS (NEON) ---
    try:
        # Borrar Logs Viejos
        query_logs = delete(AuditLog).where(AuditLog.timestamp < thirty_days_ago)
        result_logs = session.exec(query_logs)
        deleted_logs = result_logs.rowcount

        # Borrar Invitaciones Usadas y Viejas
        query_invites = delete(InviteCode).where(
            InviteCode.is_used == True, InviteCode.created_at < thirty_days_ago
        )
        result_invites = session.exec(query_invites)
        deleted_invites = result_invites.rowcount

        session.commit()
    except Exception as e:
        session.rollback()
        logger.error(f"❌ [GARBAGE_COLLECTOR] Fallo crítico en la DB: {e}")
        raise HTTPException(status_code=500, detail="Fallo en la purga de PostgreSQL")

    # --- 2. LIMPIEZA DE INFRAESTRUCTURA (CLOUDFLARE R2) ---
    deleted_temp_files = 0
    try:
        # Listamos objetos en la carpeta temp/
        response = s3_client.list_objects_v2(
            Bucket=settings.R2_BUCKET_NAME, Prefix="temp/"
        )

        if "Contents" in response:
            objects_to_delete = []
            for obj in response["Contents"]:
                # Si el archivo lleva más de 24h sin ser movido a /prompts, es basura.
                if obj["LastModified"] < one_day_ago:
                    objects_to_delete.append({"Key": obj["Key"]})

            if objects_to_delete:
                # La API de S3/R2 permite borrar hasta 1000 objetos por petición HTTP
                for i in range(0, len(objects_to_delete), 1000):
                    batch = objects_to_delete[i : i + 1000]
                    s3_client.delete_objects(
                        Bucket=settings.R2_BUCKET_NAME, Delete={"Objects": batch}
                    )
                deleted_temp_files = len(objects_to_delete)

    except Exception as e:
        logger.error(f"❌ [GARBAGE_COLLECTOR] Fallo crítico en R2: {e}")
        # No lanzamos excepción aquí para que al menos la limpieza de DB sea exitosa
        pass

    logger.info(
        f"✅ [GARBAGE_COLLECTOR] Completado. Logs: {deleted_logs} | Invites: {deleted_invites} | R2_Temp: {deleted_temp_files}"
    )

    return {
        "status": "success",
        "deleted_logs": deleted_logs,
        "deleted_invites": deleted_invites,
        "deleted_temp_files": deleted_temp_files,
    }

@router.delete("/users/{user_id}")
def delete_user_completely(
    user_id: str,
    background_tasks: BackgroundTasks,
    current_admin: User = Depends(get_current_admin),
    session: Session = Depends(get_session),
):
    """
    Exterminio Total en Cascada.
    Respeta la integridad referencial de PostgreSQL (Foreign Keys).
    """
    target_user = session.get(User, user_id)
    if not target_user:
        raise HTTPException(status_code=404, detail="Usuario no encontrado")

    if target_user.role == "admin":
        raise HTTPException(
            status_code=403, detail="No se puede borrar a un administrador."
        )

    try:
        # 1. Recuperar todas las URLs de imágenes antes de borrar la DB
        prompts = session.exec(
            select(ImagePrompt).where(ImagePrompt.user_id == target_user.id)
        ).all()
        image_urls_to_delete = [p.image_url for p in prompts if p.image_url]

        # 2. Delegar borrado físico de R2 a Background Task
        if image_urls_to_delete:
            background_tasks.add_task(purge_r2_images_batch, image_urls_to_delete)

        # 3. 🟢 RESOLVER CONSTRAINT: Desvincular el Código de Invitación
        # No borramos el código para que siga constando como "usado", pero rompemos el enlace.
        invites = session.exec(
            select(InviteCode).where(InviteCode.used_by_id == target_user.id)
        ).all()
        for inv in invites:
            inv.used_by_id = None
            session.add(inv)

        # 4. 🟢 BORRADO EN CASCADA (Desde los hijos hasta el padre)
        # Borramos las métricas de galería que pertenezcan a él
        session.exec(
            delete(GalleryMetric).where(GalleryMetric.user_id == target_user.id)
        )

        # Borramos las imágenes
        session.exec(delete(ImagePrompt).where(ImagePrompt.user_id == target_user.id))

        # Borramos las colecciones
        session.exec(delete(Directory).where(Directory.user_id == target_user.id))

        # Finalmente, ejecutamos al usuario
        session.delete(target_user)

        # Consolidamos todos los cambios en una sola transacción ACID
        session.commit()

        AuditService.log_action(
            session,
            current_admin.username,
            "USER_DELETED",
            f"Exterminado: @{target_user.username} y {len(prompts)} prompts.",
        )

        return {
            "status": "success",
            "message": f"Usuario @{target_user.username} exterminado.",
        }

    except Exception as e:
        session.rollback()
        logger.error(f"Error crítico en exterminio de usuario: {e}")
        raise HTTPException(
            status_code=500, detail="Fallo al ejecutar el exterminio en cascada."
        )

def purge_r2_images_batch(image_urls: list):
    """Borra múltiples archivos de R2 en lote"""
    from app.services.storage import s3_client
    from app.core.config import settings

    base_url = settings.R2_PUBLIC_URL.rstrip("/")

    objects_to_delete = []
    for url in image_urls:
        if url.startswith(base_url):
            key = url.replace(base_url + "/", "")
            objects_to_delete.append({"Key": key})

    if objects_to_delete:
        for i in range(0, len(objects_to_delete), 1000):
            batch = objects_to_delete[i : i + 1000]
            try:
                s3_client.delete_objects(
                    Bucket=settings.R2_BUCKET_NAME, Delete={"Objects": batch}
                )
            except Exception as e:
                pass  # Fallo silencioso en background


@router.post("/storage/purge-all")
def purge_entire_r2_bucket(
    current_admin: User = Depends(get_current_admin),
    session: Session = Depends(get_session),
):
    """
    ⚠️ BOTÓN DEL PÁNICO: Borra TODOS los archivos del bucket R2 sin piedad.
    Útil para entornos de desarrollo o tras un reset completo de la base de datos.
    """
    logger.warning(
        f"☢️ [R2_PURGE] Protocolo de aniquilación iniciado por: {current_admin.username}"
    )

    deleted_count = 0
    try:
        # Paginación (S3 devuelve resultados en lotes de máximo 1000)
        paginator = s3_client.get_paginator("list_objects_v2")
        pages = paginator.paginate(Bucket=settings.R2_BUCKET_NAME)

        for page in pages:
            if "Contents" in page:
                objects_to_delete = [{"Key": obj["Key"]} for obj in page["Contents"]]

                # Borramos el lote actual
                if objects_to_delete:
                    response = s3_client.delete_objects(
                        Bucket=settings.R2_BUCKET_NAME,
                        Delete={"Objects": objects_to_delete},
                    )
                    deleted_count += len(response.get("Deleted", []))

        logger.info(
            f"✅ [R2_PURGE] Exterminio completado. Archivos borrados: {deleted_count}"
        )
        return {
            "status": "success",
            "message": f"Bucket purgado con éxito. Se borraron {deleted_count} archivos.",
        }

    except Exception as e:
        logger.error(f"❌ [R2_PURGE] Fallo crítico durante la aniquilación: {e}")
        raise HTTPException(
            status_code=500, detail=f"Fallo al purgar Cloudflare R2: {str(e)}"
        )

@router.get("/single/{image_key}")
def descargar_una_imagen(
    image_key: str,
    current_user: User = Depends(get_current_user),
    session: Session = Depends(get_session),
):
    """Resolución sincrónica instantánea para archivos únicos."""

    statement = select(ImagePrompt).where(ImagePrompt.image_url.contains(image_key))
    prompt = session.exec(statement).first()

    if not prompt:
        raise HTTPException(
            status_code=404, detail="Imagen no encontrada en la base de datos."
        )

    if prompt.directory_id:
        col = session.get(Directory, prompt.directory_id)
        if col and col.is_private and col.user_id != current_user.id:
            raise HTTPException(
                status_code=403, detail="No tienes permiso. Esta imagen es privada."

@router.get("/studio")
def buscar_en_studio(
    q: str = Query(..., description="Texto a buscar semánticamente"),
    limit: int = 20,
    current_user: User = Depends(get_current_user),
    session: Session = Depends(get_session),
):
    """
    🔍 Buscador Semántico Local (Studio):
    1. Convierte la query en un vector.
    2. Busca en la DB los prompts que más se acerquen a ese significado.
    """
    if not q:
        return {"status": "success", "query": q, "results": []}

    query_vector = EmbeddingService.generate_vector(q)

    if not query_vector:
        return {"status": "error", "message": "No se pudo procesar la búsqueda"}

    statement = (
        select(ImagePrompt)
        .where(ImagePrompt.user_id == current_user.id)
        .order_by(ImagePrompt.embedding.cosine_distance(query_vector))
        .limit(limit)
    )

    resultados = session.exec(statement).all()

    for r in resultados:
        r.embedding = []

    return {"status": "success", "query": q, "results": resultados}

import os
import jwt
import uuid
import bcrypt
from datetime import datetime, timedelta
from dotenv import load_dotenv

# Cargamos el .env
load_dotenv()

# Configuraciones del Token
SECRET_KEY = os.getenv("SECRET_KEY", "super_clave_secreta_de_respaldo")
ALGORITHM = "HS256"

# 🟢 Las nuevas duraciones de los tokens
ACCESS_TOKEN_EXPIRE_MINUTES = 15
REFRESH_TOKEN_EXPIRE_DAYS = 7


def get_password_hash(password: str) -> str:
    pwd_bytes = password.encode("utf-8")
    salt = bcrypt.gensalt()
    hashed_password = bcrypt.hashpw(pwd_bytes, salt)
    return hashed_password.decode("utf-8")


def verify_password(plain_password: str, hashed_password: str) -> bool:
    pwd_bytes = plain_password.encode("utf-8")
    hash_bytes = hashed_password.encode("utf-8")
    return bcrypt.checkpw(pwd_bytes, hash_bytes)


def create_access_token(data: dict) -> str:
    """Fabrica el Pase VIP (JWT) de 15 minutos"""
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
    to_encode.update({"exp": expire, "type": "access"})

    encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)
    return encoded_jwt


def create_refresh_token(data: dict, family_id: str = None) -> dict:
    """Fabrica el Pase Maestro Mutante de 7 días"""
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(days=REFRESH_TOKEN_EXPIRE_DAYS)

    # Generamos el ADN del token
    token_id = str(uuid.uuid4())
    token_family = family_id if family_id else str(uuid.uuid4())

    to_encode.update(
        {
            "exp": expire,
            "type": "refresh",
            "jti": token_id,
            "family": token_family,
        }
    )

    encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)

    # Devolvemos el JWT y sus IDs para que el router los guarde en Redis
    return {"token": encoded_jwt, "jti": token_id, "family_id": token_family}

import logging
from contextlib import contextmanager

from selenium import webdriver
from selenium.webdriver.chrome.options import Options

logger = logging.getLogger("webdriver")


def _get_chrome_options():
    """Genera las opciones estrictas para que Chrome funcione dentro de Docker."""
    options = Options()
    options.add_argument("--headless=new")
    options.add_argument("--no-sandbox")
    options.add_argument("--disable-dev-shm-usage")
    options.add_argument("--disable-gpu")
    options.add_argument(
        "user-agent=Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
    )
    return options


@contextmanager
def navegador_seguro():
    """
    Context manager que levanta Chrome y garantiza su destrucción al terminar.
    Uso: with navegador_seguro() as driver: ...
    """
    logger.info("🌐 Levantando instancia de Chrome Headless...")
    driver = webdriver.Chrome(options=_get_chrome_options())
    try:

import logging

from selenium.webdriver.common.by import By
from selenium.webdriver.support import expected_conditions as EC
from selenium.webdriver.support.ui import WebDriverWait

logger = logging.getLogger("analytics")

def extraer_metadata_galeria(driver, url: str, thumb_url: str) -> dict:
    """Extrae metadata y asocia la miniatura usando un driver existente."""
    # Abrimos una nueva pestaña para no perder la página de búsqueda
    driver.execute_script("window.open(arguments[0], '_blank');", url)
    driver.switch_to.window(driver.window_handles[-1])
    
    data = {"url": url, "thumb": thumb_url, "tags": {}, "title": "N/A", "pages": 0}
    wait = WebDriverWait(driver, 10)
    
    try:
        wait.until(EC.presence_of_element_located((By.ID, "gn")))
        data["title"] = driver.find_element(By.ID, "gn").text
        
        for row in driver.find_elements(By.CSS_SELECTOR, "#gdd tr"):
            if "Number of Pages" in row.text:
                data["pages"] = row.find_element(By.CLASS_NAME, "gdt2").text.split()[0]

        for row in driver.find_elements(By.CSS_SELECTOR, "#taglist tr"):
            cat = row.find_element(By.CLASS_NAME, "tc").text.replace(":", "").lower()
            tags = [a.text.lower() for a in row.find_elements(By.CSS_SELECTOR, "div a")]
            data["tags"][cat] = tags
    except Exception as e:
        logger.warning(f"⚠️ Falló la extracción en la galería (Posible aviso +18): {e}")
    finally:
        driver.close() # Cerramos la pestaña
        driver.switch_to.window(driver.window_handles[0]) # Volvems a la principal
    
    return data

def generar_resumen_analitico(db: list):
    """Genera un reporte analítico de los tags y artistas más populares."""
    all_artists = []
    all_tags = []
    for item in db:
        if "artist" in item.get("tags", {}):
            all_artists.extend(item["tags"]["artist"])
        for cat, tags in item.get("tags", {}).items():
            if cat != "artist": 
                all_tags.extend(tags)
                
    resumen = f"\n{'='*40}\n📊 RESUMEN ANALÍTICO\n{'='*40}\n"
    resumen += f"✅ Total Galerías analizadas: {len(db)}\n\n🎨 Top 5 Artistas:\n"
    for art, count in Counter(all_artists).most_common(5):
        resumen += f"   - {art.capitalize()}: {count} galerías\n"
        
    resumen += "\n🏷️ Top 5 Tags Relacionados:\n"
    for tag, count in Counter(all_tags).most_common(5):
        resumen += f"   - {tag}: {count} coincidencias\n"
    resumen += f"{'='*40}\n"
    
    logger.info(resumen)

import os

import logging
import requests
from app.core.config import settings
from sqlalchemy import or_, func
from google.oauth2 import id_token
from google.auth.transport import requests as google_requests

GOOGLE_CLIENT_ID = os.getenv("GOOGLE_CLIENT_ID")

import jwt
from fastapi import HTTPException
from sqlmodel import Session, select
from app.utils.username import normalize_username
from app.models.user import User, InviteCode
from app.core.security import (

    ALGORITHM,

class AuthService:

    @staticmethod
    def register_user(user_data, session: Session):
        logger = logging.getLogger("auth_service")
        logger.info(f"📝 [AUTH_SERVICE] Intento de registro. Email: {user_data.email}")

        # 👑 1. LÓGICA DEL CREADOR (Bypass del Primer Usuario)
        user_count = session.exec(select(func.count(User.id))).one()
        is_first_user = user_count == 0

        invite = None

        if is_first_user:
            assigned_role = "admin"
            logger.info("👑 Primer usuario detectado (Local). Asignando ADMIN.")
        else:
            # 1b. Validación de código para usuarios normales
            logger.info(
                f"🔑 [AUTH_SERVICE] Verificando código de invitación: {user_data.invite_code}"
            )
            statement = (
                select(InviteCode)
                .where(InviteCode.code == user_data.invite_code)
                .where(InviteCode.is_used == False)
            )

            invite = session.exec(statement).first()

            if not invite:
                raise HTTPException(
                    status_code=400, detail="Código de invitación inválido o ya usado"
                )
            assigned_role = "admin" if invite.code.startswith("TROYA-") else "voyager"

        # 2. Validación de usuario existente
        email_exists = session.exec(
            select(User).where(User.email == user_data.email)
        ).first()

        if email_exists:
            raise HTTPException(
                status_code=400, detail="Este correo electrónico ya está registrado."
            )

        base_username = re.sub(r"[^a-z0-9]", "", user_data.email.split("@")[0].lower())
        if not base_username:
            base_username = "user"

        username = base_username
        counter = 1

        while session.exec(select(User).where(User.username == username)).first():
            username = f"{base_username}{counter}"
            counter += 1

        # 4. Creación
        new_user = User(
            email=user_data.email,
            username=username,
            hashed_password=get_password_hash(user_data.password),
            role=assigned_role,
        )

        session.add(new_user)
        session.commit()
        session.refresh(new_user)

        # 5. Quema de código SOLO si usó uno
        if invite:
            invite.is_used = True
            invite.used_by_id = new_user.id
            session.add(invite)
            session.commit()

        logger.info(
            f"✨ ¡Usuario registrado! Email: {new_user.email} -> Username asignado: @{new_user.username}"
        )
        return new_user

    @staticmethod
    def update_username(
        current_user: User, new_username_raw: str, session: Session, redis_client
    ):
        new_username = normalize_username(new_username_raw)

        if not new_username or len(new_username) < 3:
            raise HTTPException(status_code=400, detail="Nombre inválido")
        if new_username == current_user.username:
            return new_username, None, None

        exists = session.exec(select(User).where(User.username == new_username)).first()
        if exists:
            raise HTTPException(status_code=400, detail="El nombre ya está en uso")

        old_username = current_user.username
        keys = redis_client.keys(f"session:{old_username}:*")
        if keys:
            redis_client.delete(*keys)

        current_user.username = new_username
        session.add(current_user)
        session.commit()
        session.refresh(current_user)

        new_access_token = create_access_token(
            data={"sub": new_username, "role": current_user.role}
        )
        new_refresh_data = create_refresh_token(data={"sub": new_username})

        redis_key = f"session:{new_username}:{new_refresh_data['family_id']}"
        redis_client.setex(redis_key, 60 * 60 * 24 * 7, new_refresh_data["jti"])

        return new_username, new_access_token, new_refresh_data["token"]

    @staticmethod
    def change_password(data, current_user: User, session: Session):
        """Lógica para cambiar contraseñas."""
        if not verify_password(data.old_password, current_user.hashed_password):
            raise HTTPException(
                status_code=400, detail="La contraseña actual es incorrecta"
            )

        current_user.hashed_password = get_password_hash(data.new_password)
        session.add(current_user)
        session.commit()

    @staticmethod
    def get_active_sessions(username: str, redis_client):
        """Consulta Redis para obtener sesiones de un usuario."""
        prefix = f"session:{username}:*"
        keys = redis_client.keys(prefix)
        sessions = []
        for key in keys:
            family_id = key.split(":")[-1]
            sessions.append(
                {"family_id": family_id, "last_active_jti": redis_client.get(key)}
            )
        return sessions

    @staticmethod
    def revoke_all_sessions(username: str, redis_client):
        """Botón del pánico: borra todas las sesiones de Redis."""
        prefix = f"session:{username}:*"
        keys = redis_client.keys(prefix)
        if keys:
            redis_client.delete(*keys)

    @staticmethod
    def authenticate_user(user_data, session: Session, redis_client):
        logger = logging.getLogger("auth_service")
        logger.info(f"🔐 [AUTH] Intentando login local para: {user_data.username}")

        user = session.exec(
            select(User).where(User.username == user_data.username)
        ).first()

        if not user:
            logger.warning(
                f"❌ [AUTH] Usuario no encontrado en DB: {user_data.username}"
            )
            raise HTTPException(
                status_code=401, detail="Usuario o contraseña incorrectos"
            )

        if not verify_password(user_data.password, user.hashed_password):
            logger.warning(
                f"❌ [AUTH] Contraseña incorrecta para: {user_data.username}"
            )
            raise HTTPException(
                status_code=401, detail="Usuario o contraseña incorrectos"
            )

        logger.info(
            f"✅ [AUTH] Password verificado. Generando tokens para @{user.username}"
        )

        try:
            access_token = create_access_token(
                data={"sub": user.username, "role": user.role}
            )
            refresh_data = create_refresh_token(data={"sub": user.username})

            redis_key = f"session:{user.username}:{refresh_data['family_id']}"
            redis_client.setex(redis_key, 60 * 60 * 24 * 7, refresh_data["jti"])

            logger.info(f"🚀 [AUTH] Sesión registrada en Redis. Key: {redis_key}")
            return access_token, refresh_data["token"], user.username
        except Exception as e:
            logger.error(f"💥 [AUTH] Error crítico en el registro de sesión: {str(e)}")
            raise e

    @staticmethod
    def rotate_tokens(refresh_token: str, session: Session, redis_client):
        logger = logging.getLogger("auth_service")
        logger.info("🔄 [AUTH_SERVICE] Iniciando secuencia de rotación de tokens...")

        if not refresh_token:
            logger.warning(
                "🚫 [AUTH_SERVICE] Rotación fallida: No se recibió token de refresco."
            )
            raise HTTPException(status_code=401, detail="No hay token de refresco")

        try:
            payload = jwt.decode(refresh_token, SECRET_KEY, algorithms=[ALGORITHM])
            username, jti, family_id = (
                payload.get("sub"),
                payload.get("jti"),
                payload.get("family"),
            )

            logger.info(
                f"🔍 [AUTH_SERVICE] Token decodificado -> User: {username} | JTI: {jti} | Family: {family_id}"
            )

            if not all([username, jti, family_id]) or payload.get("type") != "refresh":
                logger.error(
                    "❌ [AUTH_SERVICE] Rotación fallida: Estructura del token inválida o no es tipo 'refresh'."
                )
                raise HTTPException(
                    status_code=401, detail="Estructura de token inválida"
                )

            redis_key = f"session:{username}:{family_id}"
            active_jti = redis_client.get(redis_key)

            logger.info(
                f"📡 [AUTH_SERVICE] Consultando Redis -> Clave: {redis_key} | JTI Guardado: {active_jti}"
            )

            if active_jti and active_jti != jti:
                logger.warning(
                    f"🚨 [AUTH_SERVICE] ALERTA DE SEGURIDAD (Posible Robo o Doble Petición)."
                )
                logger.warning(f"   -> JTI Recibido: {jti}")
                logger.warning(f"   -> JTI en Redis: {active_jti}")
                redis_client.delete(redis_key)
                logger.info(
                    f"🧹 [AUTH_SERVICE] Sesión {family_id} destruida en Redis por seguridad."
                )
                raise HTTPException(
                    status_code=403, detail="ALERTA: Robo de sesión detectado."
                )

            if not active_jti:
                logger.warning(
                    f"⚠️ [AUTH_SERVICE] Rotación rechazada: La sesión {family_id} ya expiró o fue cerrada remotamente."
                )
                raise HTTPException(
                    status_code=401, detail="Sesión expirada o cerrada remotamente."
                )

            user = session.exec(select(User).where(User.username == username)).first()
            if not user:
                logger.error(
                    f"❌ [AUTH_SERVICE] El usuario {username} existe en el token pero no en la Base de Datos."
                )
                raise HTTPException(status_code=401, detail="Usuario no encontrado")

            # Generación de nuevos tokens
            new_access_token = create_access_token(
                data={"sub": username, "role": user.role}
            )
            new_refresh_data = create_refresh_token(
                data={"sub": username}, family_id=family_id
            )

            # Actualizamos Redis
            redis_client.setex(redis_key, 60 * 60 * 24 * 7, new_refresh_data["jti"])

            logger.info(
                f"✅ [AUTH_SERVICE] Rotación EXITOSA. Nuevo JTI asignado: {new_refresh_data['jti']}"
            )

            return new_access_token, new_refresh_data["token"]

        except jwt.ExpiredSignatureError:
            logger.warning("⏱️ [AUTH_SERVICE] El Refresh Token ha caducado por tiempo.")
            raise HTTPException(
                status_code=401, detail="Token expirado. Inicia sesión."
            )
        except jwt.InvalidTokenError as e:
            logger.error(f"❌ [AUTH_SERVICE] Error de firma JWT: {str(e)}")
            raise HTTPException(status_code=401, detail="Firma inválida.")

    @staticmethod
    def verify_google_token(token: str):
        """Verifica criptográficamente el JWT contra los servidores de Google"""
        if not GOOGLE_CLIENT_ID:
            raise HTTPException(
                status_code=500,
                detail="Falta configurar GOOGLE_CLIENT_ID en el servidor",
            )
        try:
            idinfo = id_token.verify_oauth2_token(
                token, google_requests.Request(), GOOGLE_CLIENT_ID
            )
            return idinfo

import uuid

from pydantic import BaseModel


class DirectoryCreate(BaseModel):
    name: str
    parent_id: Optional[uuid.UUID] = None


class DirectoryMove(BaseModel):
    parent_id: Optional[uuid.UUID] = None


class DirectoryService:
    @staticmethod
    def create_directory(
        session: Session, user: User, data: DirectoryCreate
    ) -> Directory:
        existing = session.exec(
            select(Directory).where(
                Directory.user_id == user.id,
                Directory.parent_id == data.parent_id,
                Directory.name == data.name,
            )
        ).first()

        if existing:
            raise HTTPException(
                status_code=400, detail="Ya existe una carpeta con ese nombre."
            )

        clean_name = re.sub(r"[^a-z0-9]+", "-", data.name.lower().strip())
        unique_slug = f"{clean_name.strip('-')}-{uuid.uuid4().hex[:6]}"

        nuevo_dir = Directory(
            user_id=user.id,
            name=data.name,
            parent_id=data.parent_id,
            slug=unique_slug,
        )
        session.add(nuevo_dir)
        session.commit()
        session.refresh(nuevo_dir)
        return nuevo_dir

    @staticmethod
    def get_directories(session: Session, user: User) -> List[Directory]:
        return session.exec(select(Directory).where(Directory.user_id == user.id)).all()

    @staticmethod
    def delete_directory(session: Session, user: User, directory_id: uuid.UUID):
        dir_to_delete = session.exec(
            select(Directory).where(
                Directory.id == directory_id, Directory.user_id == user.id
            )
        ).first()

        if not dir_to_delete:
            raise HTTPException(status_code=404, detail="Carpeta no encontrada")

        try:
            session.delete(dir_to_delete)
            session.commit()
            return {"status": "success", "message": "Carpeta eliminada"}

        except IntegrityError:
            session.rollback()
            raise HTTPException(status_code=409, detail="DIRECTORY_NOT_EMPTY")

    @staticmethod
    def rename_directory(
        session: Session, user: User, directory_id: uuid.UUID, new_name: str
    ):
        directory = session.get(Directory, directory_id)
        if not directory or directory.user_id != user.id:
            raise HTTPException(status_code=404, detail="Directorio no encontrado")

        if new_name:
            directory.name = new_name
            session.add(directory)
            session.commit()

        return {"status": "success", "message": "Renombrado con éxito"}

    @staticmethod
    def move_directory(
        session: Session, user: User, directory_id: uuid.UUID, data: DirectoryMove
    ):
        dir_to_move = session.exec(
            select(Directory).where(
                Directory.id == directory_id, Directory.user_id == user.id
            )
        ).first()

        if not dir_to_move:
            raise HTTPException(status_code=404, detail="Carpeta no encontrada")

        if data.parent_id == directory_id:
            raise HTTPException(
                status_code=400, detail="No puedes mover una carpeta dentro de sí misma"
            )

        if data.parent_id:
            target_dir = session.exec(
                select(Directory).where(
                    Directory.id == data.parent_id, Directory.user_id == user.id
                )
            ).first()
            if not target_dir:
                raise HTTPException(
                    status_code=404, detail="Carpeta destino no encontrada"
                )

        dir_to_move.parent_id = data.parent_id
        session.add(dir_to_move)
        session.commit()
        session.refresh(dir_to_move)

        return {"status": "success", "message": "Carpeta movida", "data": dir_to_move}

def generar_zip_coleccion(directory_id: str, user_email: str):
    """
    Tarea asíncrona para RQ. Empaqueta una colección entera en un ZIP
    usando buffers de memoria y la sube de vuelta a R2 como un enlace temporal.
    """
    s3_client = boto3.client(
        "s3",
        endpoint_url=settings.R2_ENDPOINT_URL,
        aws_access_key_id=settings.R2_ACCESS_KEY_ID,
        aws_secret_access_key=settings.R2_SECRET_ACCESS_KEY,
        region_name="auto",
    )

    with Session(engine) as session:
        coleccion = session.get(Directory, directory_id)
        if not coleccion:
            return

        prompts = session.exec(
            select(ImagePrompt).where(ImagePrompt.directory_id == directory_id)
        ).all()

    # Buffer en memoria para no usar el disco SSD del servidor
    zip_buffer = io.BytesIO()
    metadata_list = []

    with zipfile.ZipFile(zip_buffer, "w", zipfile.ZIP_DEFLATED) as zip_file:
        for idx, prompt in enumerate(prompts):
            # 1. Extraer key de R2 desde la URL pública
            file_key = prompt.image_url.split("/")[-1]
            if "prompts/" not in file_key:
                file_key = f"prompts/{file_key}"

            # 2. Descargar imagen desde R2 a memoria
            try:
                response = s3_client.get_object(
                    Bucket=settings.R2_BUCKET_NAME, Key=file_key
                )
                image_bytes = response["Body"].read()

                # 3. Escribir imagen en el ZIP
                filename = f"{idx:03d}_{prompt.title.replace(' ', '_')}.png"
                zip_file.writestr(filename, image_bytes)

                # 4. Guardar metadatos para el JSON final
                metadata_list.append(
                    {
                        "filename": filename,
                        "prompt": prompt.prompt,
                        "model": prompt.model,
                        "sampler": prompt.sampler,
                    }
                )
            except Exception:
                continue  # Tolerancia a fallos: si una imagen falla, el ZIP sigue

        # 5. Escribir el manifiesto JSON con todos los prompts
        zip_file.writestr("prompts_metadata.json", json.dumps(metadata_list, indent=2))

    zip_buffer.seek(0)

    # 6. Subir el ZIP temporal a R2
    zip_key = f"temp/downloads/{directory_id}.zip"
    s3_client.upload_fileobj(
        zip_buffer,
        settings.R2_BUCKET_NAME,
        zip_key,
        ExtraArgs={"ContentType": "application/zip"},
    )

    # 7. Generar URL pre-firmada (caduca en 1 hora)
    download_url = s3_client.generate_presigned_url(
        "get_object",
        Params={"Bucket": settings.R2_BUCKET_NAME, "Key": zip_key},
        ExpiresIn=3600,
    )

    # (Opcional) Aquí notificarías al usuario vía WebSockets o Email de que su ZIP está listo
    return download_url

import logging
import threading  # 👈 IMPORTANTE: Importamos el gestor de candados
from fastembed import TextEmbedding

logger = logging.getLogger(__name__)


class EmbeddingService:
    _model = None
    _lock = threading.Lock()  # 👈 Creamos un candado para los hilos

    @classmethod
    def get_model(cls):
        if cls._model is None:
            with cls._lock:
                if cls._model is None:
                    logger.info(
                        "🧠 Inicializando modelo FastEmbed en memoria global (Solo 1 vez)..."
                    )
                    cls._model = TextEmbedding(model_name="BAAI/bge-small-en-v1.5")
        return cls._model

    @classmethod
    def generate_vector(cls, texto: str) -> list[float]:
        try:
            if not texto:
                return [0.0] * 384

            model = cls.get_model()

            embeddings = list(model.embed([texto]))
            return embeddings[0].tolist()

        except Exception as e:
            logger.error(f"❌ Error generando vector para '{texto}': {e}")
            return [0.0] * 384

class PromptService:
    @staticmethod
    def process_quarantined_upload(
        session: Session,
        current_user: "User",  # Type hint genérico si no lo importas directamente
        data: Any,
    ) -> "ImagePrompt":

        # =================================================================
        # 🛡️ 1. ESCUDO ANTI PATH-TRAVERSAL Y SSRF (EARLY RETURN / FAIL FAST)
        # =================================================================
        # Decodificamos la URL recibida para atrapar intentos de ofuscación
        # Ej: '%2e%2e%2f' se decodifica como '../'
        decoded_url = urllib.parse.unquote(data.image_url).lower()

        if ".." in decoded_url or "../" in decoded_url:
            logger.warning(
                f"🚨 [SECURITY] Intento de Path Traversal detectado por el usuario {current_user.id}: {data.image_url}"
            )
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Intento de evasión de directorio detectado (Path Traversal).",
            )

        # =================================================================
        # 📂 2. PROCESAMIENTO Y VALIDACIÓN DE RUTAS
        # =================================================================
        base_url = settings.R2_PUBLIC_URL.rstrip("/") + "/"
        temp_key = data.image_url.replace(base_url, "")

        if not temp_key.startswith(f"temp/{current_user.id}/"):
            logger.warning(
                f"🚨 [SECURITY] Ruta de cuarentena inválida. Esperado: temp/{current_user.id}/, Recibido: {temp_key}"
            )
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Ruta de cuarentena inválida o pertenece a otro usuario.",
            )

        # =================================================================
        # 🔍 3. LECTURA DE METADATOS Y TAMAÑO
        # =================================================================
        try:
            head = s3_client.head_object(Bucket=settings.R2_BUCKET_NAME, Key=temp_key)
            size_bytes = head["ContentLength"]
            size_kb = size_bytes / 1024
            logger.info(
                f"📥 [QUARANTINE-IN] Recibido desde Temp: {temp_key} | Tamaño: {size_kb:.2f} KB"
            )
        except Exception as e:
            logger.error(
                f"❌ [QUARANTINE-ERROR] No se pudo leer el head_object de {temp_key}: {e}"
            )
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail="Archivo temporal no encontrado en el almacenamiento.",
            )

        # =================================================================
        # 🛡️ 4. VALIDACIÓN DE BYTES MÁGICOS (SEGURIDAD DE ARCHIVOS)
        # =================================================================
        try:
            # 🟢 Aumentamos el rango a 8KB para pillar todos los metadatos JPEG/WebP
            response = s3_client.get_object(
                Bucket=settings.R2_BUCKET_NAME,
                Key=temp_key,
                Range="bytes=0-8192",
            )
            file_bytes = response["Body"].read()
        except Exception as e:
            logger.error(f"❌ [QUARANTINE-ERROR] Fallo al descargar bytes: {e}")
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail="Error leyendo el contenido del archivo temporal.",
            )

        if not validate_magic_bytes(file_bytes):
            delete_file(temp_key)  # 🔥 Destruir archivo corrupto o malicioso
            logger.warning(
                f"🚨 [SECURITY] Magic bytes inválidos detectados. Archivo destruido: {temp_key}"
            )
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="El archivo está corrupto o es malicioso.",
            )

        # =================================================================
        # 📐 5. EXTRACCIÓN DE DIMENSIONES (PIL)
        # =================================================================
        width, height = None, None
        try:
            with Image.open(BytesIO(file_bytes)) as img:
                width, height = img.size
                logger.info(
                    f"📏 [QUARANTINE-SIZE] Imagen detectada en Backend: {width}x{height}"
                )
        except Exception as e:
            logger.warning(
                f"⚠️ [QUARANTINE-WARN] PIL no pudo leer dimensiones de los 8KB: {e}"
            )
            pass

        if not width and data.resolution and "x" in data.resolution.lower():
            try:
                w, h = data.resolution.lower().split("x")
                width, height = int(w.strip()), int(h.strip())
            except Exception:
                pass

        # =================================================================
        # 📦 6. CONSOLIDACIÓN A ALMACENAMIENTO PERMANENTE Y DB
        # =================================================================
        permanent_key = temp_key.replace("temp/", "prompts/", 1)

        if not move_to_permanent(temp_key, permanent_key):
            logger.error(f"❌ [STORAGE-ERROR] Falló move_to_permanent para {temp_key}")
            raise HTTPException(
                status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
                detail="Error interno al consolidar el archivo.",
            )

        final_image_url = f"{base_url}{permanent_key}"

        parsed_dir_id = (
            uuid.UUID(data.directory_id)
            if data.directory_id and data.directory_id != "null"
            else None
        )

        nuevo_prompt = ImagePrompt(
            user_id=current_user.id,
            directory_id=parsed_dir_id,
            image_url=final_image_url,
            title=data.title,
            prompt=data.prompt,
            negative_prompt=data.negative_prompt,
            software=data.software,
            model=data.model,
            sampler=data.sampler,
            steps=data.steps,
            cfg_scale=data.cfg_scale,
            seed=data.seed,
            resolution=data.resolution,
            width=width,
            height=height,
            extra_data=data.extra_data,
            embedding=None,
        )

        session.add(nuevo_prompt)
        session.commit()
        session.refresh(nuevo_prompt)

        # =================================================================
        # 🤖 7. ENVÍO A COLA DE BACKGROUND (WORKERS IA)
        # =================================================================
        ai_queue.enqueue(
            process_image_ai, prompt_id=str(nuevo_prompt.id), job_timeout="5m"
        )

        logger.info(
            f"📡 [ENQUEUE] Subida individual enviada al Worker: {nuevo_prompt.id}"
        )

        return nuevo_prompt

                ImagePrompt.id.in_(prompt_ids), ImagePrompt.user_id == current_user.id
            )
        ).all()

        if not prompts_a_borrar:
            raise HTTPException(
                status_code=404, detail="No se encontraron las imágenes."
            )

        for prompt in prompts_a_borrar:
            if prompt.image_url:
                eliminar_de_r2(prompt.image_url)
            session.delete(prompt)

        session.commit()
        return len(prompts_a_borrar), [p.id for p in prompts_a_borrar]

MAGIC_SIGNATURES = {
    "png": b"\x89PNG\r\n\x1a\n",
    "jpeg": b"\xff\xd8\xff",
    "webp_riff": b"RIFF",
    "webp_webp": b"WEBP",
}


def validate_magic_bytes(file_bytes: bytes) -> bool:
    """
    Verifica la autenticidad del archivo leyendo sus primeros bytes directamente.
    """
    if not file_bytes or len(file_bytes) < 12:
        return False

    if file_bytes.startswith(MAGIC_SIGNATURES["png"]):
        return True
    if file_bytes.startswith(MAGIC_SIGNATURES["jpeg"]):
        return True
    if (
        file_bytes.startswith(MAGIC_SIGNATURES["webp_riff"])
        and file_bytes[8:12] == MAGIC_SIGNATURES["webp_webp"]
    ):
        return True

    logger.warning(
        "[SECURITY] Archivo malicioso detectado y bloqueado (Magic Bytes inválidos)."
    )
    return False

import os
import cv2
import numpy as np
import pandas as pd
import onnxruntime as rt
import urllib.request
import logging
import threading
from typing import List, Tuple
import requests

logger = logging.getLogger("wd14_service")


class WD14Tagger:
    _instance = None
    _lock = threading.Lock()

    def __new__(cls):
        with cls._lock:
            if cls._instance is None:
                cls._instance = super(WD14Tagger, cls).__new__(cls)
                cls._instance._initialize()
            return cls._instance

    def _initialize(self):
        self.model_name = "SmilingWolf/wd-swinv2-tagger-v3"
        self.model_file = "model.onnx"
        self.csv_file = "selected_tags.csv"

        base_dir = os.environ.get("STORAGE_PATH", os.path.join(os.getcwd(), ".storage"))
        self.model_dir = os.path.join(base_dir, "models", "wd14")

        os.makedirs(self.model_dir, exist_ok=True)

        self.model_path = os.path.join(self.model_dir, self.model_file)
        self.csv_path = os.path.join(self.model_dir, self.csv_file)

        self._ensure_files_exist()

        logger.info("🏷️ [WD14] Cargando CSV de etiquetas...")
        self.tags_df = pd.read_csv(self.csv_path)
        self.tag_names = self.tags_df[self.tags_df["category"].isin([0, 4])][
            "name"
        ].tolist()
        self.tag_indices = self.tags_df[
            self.tags_df["category"].isin([0, 4])
        ].index.tolist()

        logger.info("🧠 [WD14] Cargando modelo SwinV2 en ONNX Runtime (CPU)...")
        self.ort_session = rt.InferenceSession(
            self.model_path, providers=["CPUExecutionProvider"]
        )

        self.input_name = self.ort_session.get_inputs()[0].name
        self.output_name = self.ort_session.get_outputs()[0].name

        logger.info("✅ [WD14] Modelo cargado en RAM y listo para etiquetar.")

    def _ensure_files_exist(self):
        from huggingface_hub import hf_hub_download

        if not os.path.exists(self.model_path) or not os.path.exists(self.csv_path):
            logger.info(
                f"⬇️ [WD14] Descargando modelo {self.model_name} por primera vez. Esto puede tardar un poco..."
            )

            # Descargar ONNX
            hf_hub_download(
                repo_id=self.model_name,
                filename=self.model_file,
                local_dir=self.model_dir,
                local_dir_use_symlinks=False,
            )
            # Descargar CSV
            hf_hub_download(
                repo_id=self.model_name,
                filename=self.csv_file,
                local_dir=self.model_dir,
                local_dir_use_symlinks=False,
            )
            logger.info("✅ [WD14] Descarga completada.")

    def _preprocess_image(
        self, image_bytes: bytes, target_size: int = 448
    ) -> np.ndarray:
        nparr = np.frombuffer(image_bytes, np.uint8)
        img = cv2.imdecode(nparr, cv2.IMREAD_COLOR)

        if img is None:
            raise ValueError("No se pudo decodificar la imagen.")

        img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)

        h, w = img.shape[:2]
        size = max(h, w)
        pad_h, pad_w = (size - h) // 2, (size - w) // 2

        img_padded = np.full((size, size, 3), 255, dtype=np.uint8)
        img_padded[pad_h : pad_h + h, pad_w : pad_w + w] = img

        img_resized = cv2.resize(
            img_padded, (target_size, target_size), interpolation=cv2.INTER_AREA
        )

        img_float = img_resized.astype(np.float32)

        img_float = img_float[:, :, ::-1]
        img_float = np.expand_dims(img_float, axis=0)

        return img_float

    def get_tags(self, image_url: str, threshold: float = 0.35) -> List[dict]:
        try:
            response = requests.get(image_url, timeout=10)
            response.raise_for_status()

            input_tensor = self._preprocess_image(response.content)

            probs = self.ort_session.run(
                [self.output_name], {self.input_name: input_tensor}
            )[0][0]

            result_tags = []

            for i, p in enumerate(probs):
                if i in self.tag_indices and p >= threshold:
                    tag_name = self.tag_names[self.tag_indices.index(i)]
                    clean_tag = tag_name.replace("_", " ")

                    confidence = int(round(float(p) * 100))

                    result_tags.append({"name": clean_tag, "confidence": confidence})

            result_tags = sorted(
                result_tags, key=lambda x: x["confidence"], reverse=True
            )

            return result_tags

        except Exception as e:
            logger.error(f"❌ [WD14] Error etiquetando {image_url}: {e}")
            return []

class StructuredLogger:
    def __init__(self, name: str):
        self.logger = logging.getLogger(name)

    def _format(self, level, msg, request=None, extra=None):
        log_data = {
            "timestamp": datetime.utcnow().isoformat(),
            "level": level,
            "module": self.logger.name,
            "message": msg,
            "correlation_id": (
                getattr(request.state, "correlation_id", "system")
                if request
                else "system"
            ),
        }
        if extra:
            log_data.update(extra)
        return json.dumps(log_data)

    def info(self, msg, request=None, **kwargs):
        self.logger.info(self._format("INFO", msg, request, kwargs))

    def error(self, msg, request=None, **kwargs):
        self.logger.error(self._format("ERROR", msg, request, kwargs))

    def warning(self, msg, request=None, **kwargs):
        self.logger.warning(self._format("WARNING", msg, request, kwargs))

def subir_imagen_a_r2(
    file_bytes: bytes, file_name: str, content_type: str = "image/webp"
) -> str:
    """
    Sube un archivo binario a Cloudflare R2 y devuelve la URL pública final.
    """
    if not s3_client or not R2_BUCKET_NAME:
        logger.error("⚠️ Credenciales de R2 no configuradas. Abortando subida.")
        raise ValueError("Servicio de almacenamiento en la nube no configurado.")

    try:
        logger.info(f"☁️ Subiendo {file_name} a R2...")
        s3_client.put_object(
            Bucket=R2_BUCKET_NAME,
            Key=file_name,
            Body=file_bytes,
            ContentType=content_type,
            CacheControl="public, max-age=31536000, immutable",
        )

        # Construir y devolver la URL pública exacta para guardarla en PostgreSQL
        url_publica = f"{R2_PUBLIC_URL.rstrip('/')}/{file_name}"
        logger.info(f"✅ Subida exitosa: {url_publica}")
        return url_publica

    except ClientError as e:
        logger.error(f"❌ Error de conexión con R2 (Boto3): {e}")
        raise e
    except Exception as e:
        logger.error(f"❌ Error inesperado subiendo a R2: {e}")
        raise e


def generar_presigned_url(file_name: str, content_type: str) -> dict:
    """Genera un ticket de un solo uso para que el Frontend suba directo a R2"""
    if not s3_client or not R2_BUCKET_NAME:
        raise ValueError("Servicio R2 no configurado en .env.")
    try:
        # Genera una URL temporal con permiso explícito de 'put_object' (Subida)
        presigned_url = s3_client.generate_presigned_url(
            "put_object",
            Params={
                "Bucket": R2_BUCKET_NAME,
                "Key": file_name,
                "ContentType": content_type,
            },
            ExpiresIn=3600,  # El ticket caduca en 1 hora
        )

        # Esta será la URL final pública una vez subido el archivo
        url_publica = f"{R2_PUBLIC_URL.rstrip('/')}/{file_name}"
        return {"upload_url": presigned_url, "file_url": url_publica}

    except Exception as e:
        logger.error(f"❌ Error generando presigned URL: {e}")
        raise e


def eliminar_de_r2(file_url: str):
    """Borra físicamente el archivo del bucket para no dejar huérfanos"""
    if not s3_client or not R2_BUCKET_NAME or not file_url:
        return
    try:
        base_url = R2_PUBLIC_URL.rstrip("/")
        # Verificamos que la URL pertenezca a nuestro bucket
        if file_url.startswith(base_url):
            # Extraemos la ruta interna (ej. prompts/edu/archivo.png)
            key = file_url.replace(base_url + "/", "")
            logger.info(f"🗑️ Eliminando de R2: {key}")
            s3_client.delete_object(Bucket=R2_BUCKET_NAME, Key=key)
    except Exception as e:
        logger.error(f"❌ Error eliminando objeto de R2: {e}")

import unicodedata
import re


def normalize_username(username: str, max_chars: int = 30) -> str:
    if not username:
        return ""

    nfkd_form = unicodedata.normalize("NFKD", username)
    text = "".join([c for c in nfkd_form if not unicodedata.combining(c)])

    text = text.lower().strip()

    text = re.sub(r"[^a-z0-9]+", "_", text)

    text = text.strip("_")

    return text[:max_chars]

def heal_missing_dimensions():
    logger.info("🩺 Iniciando el Sanador de Dimensiones en Segundo Plano...")

    with Session(engine) as session:
        statement = select(ImagePrompt.id).where(
            or_(ImagePrompt.width == None, ImagePrompt.height == None)
        )
        ids_rotos = session.exec(statement).all()

        if not ids_rotos:
            logger.info(
                "✨ ¡Base de datos sana! Todas las imágenes tienen dimensiones."
            )
            return

        logger.info(
            f"🩹 Se han encontrado {len(ids_rotos)} imágenes sin dimensiones. Procesando..."
        )

        sanados = 0
        errores = 0

        for prompt_id in ids_rotos:
            prompt = session.get(ImagePrompt, prompt_id)
            if not prompt or not prompt.image_url:
                continue

            try:
                response = requests.get(prompt.image_url, stream=True, timeout=10)
                response.raise_for_status()

                img = Image.open(response.raw)
                width, height = img.size

                prompt.width = width
                prompt.height = height
                prompt.resolution = f"{width}x{height}"

                session.add(prompt)
                session.commit()

                sanados += 1
                logger.info(f"✅ Sanado Prompt {prompt.id}: {width}x{height}")

            except Exception as e:
                # 🟢 2. Rollback FIRST before doing anything else
                session.rollback()
                errores += 1
                # We can safely use prompt_id here, but avoid prompt.image_url if it requires lazy loading after rollback
                logger.warning(f"⚠️ Fallo al sanar ID {prompt_id}: {e}")

        logger.info(f"🏁 Ejecución completada. Sanados: {sanados} | Errores: {errores}")


if __name__ == "__main__":

# 💣 1. CONDICIÓN DE CARRERA (RACE CONDITION)
# =====================================================================
def test_race_condition_download_limits(
    client: TestClient, session: Session, normal_user: User
):
    public_col = Directory(
        user_id=normal_user.id, name="Coleccion Golosa", slug="golosa"
    )
    session.add(public_col)
    session.commit()

    login_res = client.post(
        "/api/auth/login",
        json={"username": "test_voyager", "password": "PassSegura123!"},
    )
    cookies = login_res.cookies

    def lanzar_peticion_simultanea():
        thread_client = TestClient(client.app)
        thread_client.cookies = cookies
        return thread_client.post(f"/api/download/directory/{public_col.id}")

    exitos = 0
    bloqueados = 0

    with concurrent.futures.ThreadPoolExecutor(max_workers=20) as executor:
        resultados = list(
            executor.map(lambda _: lanzar_peticion_simultanea(), range(20))
        )

    for res in resultados:
        if res.status_code in [200, 202]:
            exitos += 1
        elif res.status_code in [429, 403, 500]:
            bloqueados += 1

    assert exitos < 20, "🚨 CRÍTICO: Vulnerabilidad de Race Condition detectada."
    assert bloqueados > 0, "El escudo no actuó a tiempo."
